\section{Variation of the Action}

\subsection{Hilbert Space Approach I}

The action is defined as%
\begin{equation}
\mathcal{A}\left( \Psi ,\tau \right) =\int_{t_{i}}^{t_{f}}\mathcal{L}\left(
\Psi \left( t\right) \right) dt
\end{equation}%
where 
\begin{equation}
\mathcal{L}\left( \Psi \left( t\right) \right) =\frac{\left\langle \Psi
\left( t\right) |\left( i\frac{\partial }{\partial t}-H\right) \Psi \left(
t\right) \right\rangle }{\left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle }
\end{equation}%
the first order variation of the Lagrangian is then%
\begin{eqnarray}
\delta \mathcal{L}\left( \Psi \left( t\right) \right)  &=&\frac{1}{%
\left\langle \Psi \left( t\right) |\Psi \left( t\right) \right\rangle }%
\left\{ \left\langle \delta \Psi \left( t\right) |\left( i\frac{\partial }{%
\partial t}-H\right) \Psi \left( t\right) \right\rangle +\left\langle \Psi
\left( t\right) |\left( i\frac{\partial }{\partial t}-H\right) \delta \Psi
\left( t\right) \right\rangle \right.   \notag \\
&&\qquad \qquad \qquad \qquad \left. -\mathcal{L}\left( \Psi \left( t\right)
\right) \left\{ \left\langle \delta \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle +\left\langle \Psi \left( t\right) |\delta \Psi
\left( t\right) \right\rangle \right\} \overset{}{\left. {}\right. }\right\} 
\end{eqnarray}%
leading to the first order change in the action wrt to changes in the paths $%
\Psi \left( t\right) $ 
\begin{eqnarray}
\delta \mathcal{A}\left( \Psi ,\tau \right)  &=&\int_{t_{i}}^{t_{f}}\delta 
\mathcal{L}\left( \Psi \left( t\right) \right) dt  \notag \\
&=&\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }\left\{ \left\langle \delta \Psi \left(
t\right) |\left( i\frac{\partial }{\partial t}-H\right) \Psi \left( t\right)
\right\rangle -\mathcal{L}\left( \Psi \left( t\right) \right) \left\langle
\delta \Psi \left( t\right) |\Psi \left( t\right) \right\rangle \right.  
\notag \\
&&\qquad \qquad +\left. \left\langle \Psi \left( t\right) |\left( i\frac{%
\partial }{\partial t}-H\right) \delta \Psi \left( t\right) \right\rangle -%
\mathcal{L}\left( \Psi \left( t\right) \right) \left\langle \Psi \left(
t\right) |\delta \Psi \left( t\right) \right\rangle \right\} dt
\end{eqnarray}%
In order for the action to be stationary for \emph{arbitrary }changes $%
\left\vert \delta \Psi \left( t\right) \right\rangle $ in the paths $%
\left\vert \Psi \left( t\right) \right\rangle $ it must be stationary for
both the paths $\left\vert \delta \Psi \left( t\right) \right\rangle $ and $%
\left\vert i\delta \Psi \left( t\right) \right\rangle $ giving 
\begin{eqnarray}
&&\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }\left\{ \left\langle \delta \Psi \left(
t\right) |\left( i\frac{\partial }{\partial t}-H\right) \Psi \left( t\right)
\right\rangle -\mathcal{L}\left( \Psi \left( t\right) \right) \left\langle
\delta \Psi \left( t\right) |\Psi \left( t\right) \right\rangle \right.  
\notag \\
&&+\left. \left\langle \Psi \left( t\right) |\left( i\frac{\partial }{%
\partial t}-H\right) \delta \Psi \left( t\right) \right\rangle -\mathcal{L}%
\left( \Psi \left( t\right) \right) \left\langle \Psi \left( t\right)
|\delta \Psi \left( t\right) \right\rangle \right\} dt=0
\end{eqnarray}%
and 
\begin{eqnarray}
&&\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }\left\{ -\left\langle \delta \Psi \left(
t\right) |\left( i\frac{\partial }{\partial t}-H\right) \Psi \left( t\right)
\right\rangle +\mathcal{L}\left( \Psi \left( t\right) \right) \left\langle
\delta \Psi \left( t\right) |\Psi \left( t\right) \right\rangle \right.  
\notag \\
&&+\left. \left\langle \Psi \left( t\right) |\left( i\frac{\partial }{%
\partial t}-H\right) \delta \Psi \left( t\right) \right\rangle -\mathcal{L}%
\left( \Psi \left( t\right) \right) \left\langle \Psi \left( t\right)
|\delta \Psi \left( t\right) \right\rangle \right\} dt=0
\end{eqnarray}%
Adding and subtracting these two equations we obtain 
\begin{equation}
\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle }\left\{ \left\langle \Psi \left( t\right) |\left( i%
\frac{\partial }{\partial t}-H\right) \delta \Psi \left( t\right)
\right\rangle -\mathcal{L}\left( \Psi \left( t\right) \right) \left\langle
\Psi \left( t\right) |\delta \Psi \left( t\right) \right\rangle \right\} dt=0
\label{a1}
\end{equation}%
and 
\begin{equation}
\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle }\left\{ \left\langle \delta \Psi \left( t\right)
|\left( i\frac{\partial }{\partial t}-H\right) \Psi \left( t\right)
\right\rangle -\mathcal{L}\left( \Psi \left( t\right) \right) \left\langle
\delta \Psi \left( t\right) |\Psi \left( t\right) \right\rangle \right\} dt=0
\label{a2}
\end{equation}%
for all \emph{arbitrary }changes of path $\left\vert \delta \Psi \left(
t\right) \right\rangle $ $\lceil $Note that these equations \emph{are not }%
complex conjugate as $\mathcal{L}\left( \Psi \left( t\right) \right) \in 
\mathbb{C}$ in general$\rfloor $. The equation Eq. $\left( \ref{a1}\right) $
can be further developed to give 
\begin{equation}
\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle }\left\{ \left\langle \left[ i\frac{\partial }{%
\partial t}\right] ^{\dagger }\Psi \left( t\right) |\delta \Psi \left(
t\right) \right\rangle -\left\langle \Psi \left( t\right) |H\delta \Psi
\left( t\right) \right\rangle -\mathcal{L}\left( \Psi \left( t\right)
\right) \left\langle \Psi \left( t\right) |\delta \Psi \left( t\right)
\right\rangle \right\} dt=0
\end{equation}%
which, using appendices $\left( \ref{Appendix A}\right) $ and $\left( \ref%
{Appendix B}\right) $, gives 
\begin{eqnarray}
&&\int_{t_{i}}^{t_{f}}\left\{ i\frac{d}{dt}\left\{ \frac{\left\langle \Psi
\left( t\right) |\delta \Psi \left( t\right) \right\rangle }{\left\langle
\Psi \left( t\right) |\Psi \left( t\right) \right\rangle }\right\} +\frac{1}{%
\left\langle \Psi \left( t\right) |\Psi \left( t\right) \right\rangle }%
\left\{ \left\langle i\frac{\partial }{\partial t}\Psi \left( t\right)
|\delta \Psi \left( t\right) \right\rangle \right. \right.   \notag \\
&&\left. -\left\langle \Psi \left( t\right) |H\delta \Psi \left( t\right)
\right\rangle -\mathcal{L}\left( \Psi \left( t\right) \right) \left\langle
\Psi \left( t\right) |\delta \Psi \left( t\right) \right\rangle +\frac{%
\left\langle \Psi \left( t\right) |\delta \Psi \left( t\right) \right\rangle 
}{\left\langle \Psi \left( t\right) |\Psi \left( t\right) \right\rangle }i%
\frac{d}{dt}\ln \left\langle \Psi \left( t\right) |\Psi \left( t\right)
\right\rangle \overset{}{}\right\} dt=0
\end{eqnarray}%
leading to 
\begin{eqnarray}
&&i\left\{ \left\{ \frac{\left\langle \Psi \left( t_{f}\right) |\delta \Psi
\left( t_{f}\right) \right\rangle }{\left\langle \Psi \left( t_{f}\right)
|\Psi \left( t_{f}\right) \right\rangle }\right\} -\left\{ \frac{%
\left\langle \Psi \left( t_{i}\right) |\delta \Psi \left( t_{i}\right)
\right\rangle }{\left\langle \Psi \left( t_{i}\right) |\Psi \left(
t_{i}\right) \right\rangle }\right\} \right\} +  \notag \\
&&\qquad \int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right)
|\Psi \left( t\right) \right\rangle }\left\{ \left\langle \left( i\frac{%
\partial }{\partial t}-H\right) \Psi \left( t\right) |\delta \Psi \left(
t\right) \right\rangle \right. -  \notag \\
&&\qquad \qquad \qquad \left. \left( \mathcal{L}\left( \Psi \left( t\right)
\right) -i\frac{d}{dt}\ln \left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle \right) \left\langle \Psi \left( t\right) |\delta
\Psi \left( t\right) \right\rangle \right\} dt=0  \label{a1d}
\end{eqnarray}%
which gives if the paths have fixed endpoints and are continuous in time 
\begin{equation}
\left\langle \left( i\frac{\partial }{\partial t}-H\right) \Psi \left(
t\right) \right\vert =\left\langle \left( \mathcal{L}\left( \Psi \left(
t\right) \right) -i\frac{d}{dt}\ln \left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle \right) ^{\ast }\Psi \left( t\right)
\right\vert \qquad \forall t\in \tau 
\end{equation}%
which is equivalent to 
\begin{equation}
\left( i\frac{\partial }{\partial t}-H\right) \left\vert \Psi \left(
t\right) \right\rangle =\left( \mathcal{L}\left( \Psi \left( t\right)
\right) -i\frac{d}{dt}\ln \left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle \right) ^{\ast }\left\vert \Psi \left( t\right)
\right\rangle \qquad \forall t\in \tau 
\end{equation}%
If the paths are continuous in time equation eq. $\left( \ref{a2}\right) $
gives 
\begin{equation}
\left( i\frac{\partial }{\partial t}-H\right) \left\vert \Psi \left(
t\right) \right\rangle =\mathcal{L}\left( \Psi \left( t\right) \right)
\left\vert \Psi \left( t\right) \right\rangle \qquad \forall t\in \tau 
\end{equation}%
which implies that $\mathcal{L}\left( \Psi \left( t\right) \right) \in 
\mathbb{R}$ \emph{when} $\left\vert \Psi \left( t\right) \right\rangle $ is
a simultaneous solution to \emph{both }equations.

\subsection{Hilbert Space Approach II}

Again the action is defined as%
\begin{equation}
\mathcal{A}\left( \Psi ,\tau \right) =\int_{t_{i}}^{t_{f}}\mathcal{L}\left(
\Psi \left( t\right) \right) dt
\end{equation}%
but now 
\begin{equation}
\mathcal{L}\left( \Psi \left( t\right) \right) =\left\langle \Psi \left(
t\right) \left\vert \left( i\frac{\partial }{\partial t}-H\right) \right.
\Psi \left( t\right) \right\rangle
\end{equation}%
the first order variation of the Lagrangian is then%
\begin{equation}
\delta \mathcal{L}\left( \Psi \left( t\right) \right) =\left\langle \delta
\Psi \left( t\right) |\left( i\frac{\partial }{\partial t}-H\right) \Psi
\left( t\right) \right\rangle +\left\langle \Psi \left( t\right) |\left( i%
\frac{\partial }{\partial t}-H\right) \delta \Psi \left( t\right)
\right\rangle
\end{equation}%
In order for the action to be stationary for \emph{arbitrary }changes $%
\left\vert \delta \Psi \left( t\right) \right\rangle $ in the paths $%
\left\vert \Psi \left( t\right) \right\rangle $ it must be stationary for
both the paths $\left\vert \delta \Psi \left( t\right) \right\rangle $ and $%
\left\vert i\delta \Psi \left( t\right) \right\rangle $ giving 
\begin{equation}
\int_{t_{i}}^{t_{f}}\left\{ \left\langle \delta \Psi \left( t\right) |\left(
i\frac{\partial }{\partial t}-H\right) \Psi \left( t\right) \right\rangle
+\left\langle \Psi \left( t\right) |\left( i\frac{\partial }{\partial t}%
-H\right) \delta \Psi \left( t\right) \right\rangle \right\} dt=0
\end{equation}%
and 
\begin{equation}
\int_{t_{i}}^{t_{f}}\left\{ \left\langle \delta \Psi \left( t\right) |\left(
i\frac{\partial }{\partial t}-H\right) \Psi \left( t\right) \right\rangle
-\left\langle \Psi \left( t\right) |\left( i\frac{\partial }{\partial t}%
-H\right) \delta \Psi \left( t\right) \right\rangle \right\} dt=0
\end{equation}%
Thus for continuous time paths $\delta \Psi \left( t\right) \in \mathcal{H}%
^{N}$%
\begin{subequations}
\begin{eqnarray}
\left\langle \delta \Psi \left( t\right) |\left( i\frac{\partial }{\partial t%
}-H\right) \Psi \left( t\right) \right\rangle &=&0 \\
\left\langle \Psi \left( t\right) |\left( i\frac{\partial }{\partial t}%
-H\right) \delta \Psi \left( t\right) \right\rangle &=&0
\end{eqnarray}%
leading to 
\end{subequations}
\begin{equation}
\left( i\frac{\partial }{\partial t}-H\right) \left\vert \Psi \left(
t\right) \right\rangle =0
\end{equation}

\subsection{State Functions}

The physical states of a system are defined to be 
\begin{equation}
S=\left\{ D|D\in \mathcal{B}_{1}\left( \mathcal{H}\right) ;D\geq
0;TrD=1\right\}
\end{equation}%
(of course $S$ is a mathematical representation of the possible physical
states of a given system). A state function is a function defined on $S$ and
associates a value with a physical state and corresponds to a possible
observable of the system. Functions, $F$, defined on the cone $\mathcal{B}%
_{1+}\left( \mathcal{H}\right) $ generated by $S$ are also state functions
if 
\begin{equation}
F\left( \alpha D\right) =F\left( D\right) \quad \forall \alpha >0\text{ and }%
\forall D\in S
\end{equation}%
One can see that the Action and Lagrangian defined in Hilbert Space Approach
I are state functions while those defined in Hilbert Space Approach II are
not.

\subsection{Calculus of Variation Approach}

Equation eq. $\left( \ref{a1}\right) $ can be rewritten as 
\begin{equation}
\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle }\left\{ i\left\langle \Psi \left( t\right) |\delta 
\dot{\Psi}\left( t\right) \right\rangle -\left\langle \Psi \left( t\right)
|H\delta \Psi \left( t\right) \right\rangle -\mathcal{L}\left( \Psi \left(
t\right) \right) \left\langle \Psi \left( t\right) |\delta \Psi \left(
t\right) \right\rangle \right\} dt=0  \label{b1}
\end{equation}%
The first term of eq. $\left( \ref{b1}\right) $ can be integrated by parts 
\begin{eqnarray}
\lceil \int_{a}^{b}u(x)dv(x) &=&\left. u(x)v(x)\right\vert
_{a}^{b}-\int_{a}^{b}v(x)du(x)  \notag \\
&\shortparallel &  \notag \\
\int_{a}^{b}u(x)\frac{dv(x)}{dx}dx &=&\left. u(x)v(x)\right\vert
_{a}^{b}-\int_{a}^{b}v(x)\frac{du(x)}{dx}dx\rfloor
\end{eqnarray}%
to give 
\begin{eqnarray}
&&\int_{t_{i}}^{t_{f}}\frac{i\left\langle \Psi \left( t\right) |\delta \dot{%
\Psi}\left( t\right) \right\rangle }{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }dt  \notag \\
&=&\left. \left\{ \frac{i\left\langle \Psi \left( t\right) |\delta \Psi
\left( t\right) \right\rangle }{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }\right\} \right\vert
_{t_{i}}^{t_{f}}-\int_{t_{i}}^{t_{f}}\frac{i}{\left\langle \Psi \left(
t\right) |\Psi \left( t\right) \right\rangle }\left\{ \left\langle \dot{\Psi}%
\left( t\right) |\delta \Psi \left( t\right) \right\rangle \right.  \notag \\
&&\qquad \qquad \qquad \qquad \qquad \qquad \qquad \left. -\frac{%
\left\langle \Psi \left( t\right) |\delta \Psi \left( t\right) \right\rangle 
}{\left\langle \Psi \left( t\right) |\Psi \left( t\right) \right\rangle }%
\left\{ \left\langle \Psi \left( t\right) |\dot{\Psi}\left( t\right)
\right\rangle +\left\langle \dot{\Psi}\left( t\right) |\Psi \left( t\right)
\right\rangle \right\} \right\} dt  \notag \\
&=&\left. \left\{ \frac{i\left\langle \Psi \left( t\right) |\delta \Psi
\left( t\right) \right\rangle }{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }\right\} \right\vert
_{t_{i}}^{t_{f}}-\int_{t_{i}}^{t_{f}}\frac{i}{\left\langle \Psi \left(
t\right) |\Psi \left( t\right) \right\rangle }\left\{ \left\langle \dot{\Psi}%
\left( t\right) |\delta \Psi \left( t\right) \right\rangle \right.  \notag \\
&&\qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \left.
-\left\langle \Psi \left( t\right) |\delta \Psi \left( t\right)
\right\rangle \frac{d}{dt}\ln \left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle \overset{}{}\right\} dt
\end{eqnarray}%
leading to eq $\left( \ref{b1}\right) $ being equal to%
\begin{eqnarray}
&&\left. \left\{ \frac{i\left\langle \Psi \left( t\right) |\delta \Psi
\left( t\right) \right\rangle }{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }\right\} \right\vert _{t_{i}}^{t_{f}}  \notag
\\
&&\qquad \qquad +\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left(
t\right) |\Psi \left( t\right) \right\rangle }\left\{ \left\langle i\dot{\Psi%
}\left( t\right) |\delta \Psi \left( t\right) \right\rangle +\left\langle
\Psi \left( t\right) |\delta \Psi \left( t\right) \right\rangle i\frac{d}{dt}%
\ln \left\langle \Psi \left( t\right) |\Psi \left( t\right) \right\rangle
\right.  \notag \\
&&\qquad \qquad \qquad \qquad \qquad \qquad \qquad -\left. \left\langle \Psi
\left( t\right) |H\delta \Psi \left( t\right) \right\rangle -\mathcal{L}%
\left( \Psi \left( t\right) \right) \left\langle \Psi \left( t\right)
|\delta \Psi \left( t\right) \right\rangle \right\} dt
\end{eqnarray}%
which can be written more succinctly as 
\begin{eqnarray}
&&\left. \left\{ \frac{i\left\langle \Psi \left( t\right) |\delta \Psi
\left( t\right) \right\rangle }{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }\right\} \right\vert _{t_{i}}^{t_{f}}  \notag
\\
&&+\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }\left\{ \left\langle i\dot{\Psi}\left(
t\right) \right\vert -\left\langle \Psi \left( t\right) \right\vert
H-\left\langle \Psi \left( t\right) \right\vert \left\{ \mathcal{L}\left(
\Psi \left( t\right) \right) -i\frac{d}{dt}\ln \left\langle \Psi \left(
t\right) |\Psi \left( t\right) \right\rangle \right\} \right.  \notag \\
&&\qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad
\qquad \qquad \qquad \left. \left\vert \delta \Psi \left( t\right)
\right\rangle \overset{}{\left. {}\right. }\right\} dt=0
\end{eqnarray}%
If the end points are fixed this gives the stationarity condition 
\begin{eqnarray}
&&\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle }\left\{ \left\langle i\dot{\Psi}\left(
t\right) \right\vert -\left\langle \Psi \left( t\right) \right\vert
H-\left\langle \Psi \left( t\right) \right\vert \left\{ \mathcal{L}\left(
\Psi \left( t\right) \right) -i\frac{d}{dt}\ln \left\langle \Psi \left(
t\right) |\Psi \left( t\right) \right\rangle \right\} \right\} \left\vert
\delta \Psi \left( t\right) \right\rangle  \notag \\
&&\qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad
\qquad \qquad \qquad \qquad \qquad \qquad \qquad =0
\end{eqnarray}%
which gives 
\begin{equation}
\left\langle \left( i\frac{\partial }{\partial t}-H\right) \Psi \left(
t\right) \right\vert =\left( \mathcal{L}\left( \Psi \left( t\right) \right)
-i\frac{d}{dt}\ln \left\langle \Psi \left( t\right) |\Psi \left( t\right)
\right\rangle \right) \left\langle \Psi \left( t\right) \right\vert
\label{c1}
\end{equation}%
which is equivalent to 
\begin{equation}
\left( i\frac{\partial }{\partial t}-H\right) \left\vert \Psi \left(
t\right) \right\rangle =\left( \mathcal{L}\left( \Psi \left( t\right)
\right) -i\frac{d}{dt}\ln \left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle \right) ^{\ast }\left\vert \Psi \left( t\right)
\right\rangle
\end{equation}

Equation eq. $\left( \ref{a2}\right) $ can be cast as 
\begin{equation}
\int_{t_{i}}^{t_{f}}\frac{1}{\left\langle \Psi \left( t\right) |\Psi \left(
t\right) \right\rangle }\left\{ \left\langle \left( \left[ i\frac{\partial }{%
\partial t}\right] ^{\dagger }-H\right) \delta \Psi \left( t\right) |\Psi
\left( t\right) \right\rangle -\mathcal{L}\left( \Psi \left( t\right)
\right) \left\langle \delta \Psi \left( t\right) |\Psi \left( t\right)
\right\rangle \right\} dt=0
\end{equation}%
which after using appendices $\left( \ref{Appendix A}\right) $ and $\left( %
\ref{Appendix B}\right) $, has the form 
\begin{eqnarray}
&&i\left\{ \left\{ \frac{\left\langle \delta \Psi \left( t_{f}\right) |\Psi
\left( t_{f}\right) \right\rangle }{\left\langle \Psi \left( t_{f}\right)
|\Psi \left( t_{f}\right) \right\rangle }\right\} -\left\{ \frac{%

\subsection{Equations in Terms of Derivatives of the Lagrangian}

The first order of the variation can be expanded as 
\begin{equation*}
\delta \mathcal{A}\left( \Psi ,\tau \right) =\int_{t_{i}}^{t_{f}}\delta 
\mathcal{L}\left( \Psi \left( t\right) \right)
dt=\int_{t_{i}}^{t_{f}}\left\{ \left\langle \delta \Psi \left( t\right)
\left| \frac{\partial \mathcal{L}\left( \Psi \left( t\right) \right) }{%
\partial \Psi ^{\ast }}\right. \right\rangle +\left\langle \left. \frac{%
\partial \mathcal{L}\left( \Psi \left( t\right) \right) }{\partial \Psi }%
\right| \delta \Psi \left( t\right) \right\rangle \right\} dt
\end{equation*}%
where 
\begin{eqnarray}
\left\langle \delta \Psi \left( t\right) \left| \frac{\partial \mathcal{L}%
\left( \Psi \left( t\right) \right) }{\partial \Psi ^{\ast }}\right.
\right\rangle &=&\frac{1}{\left\langle \Psi \left( t\right) |\Psi \left(

\notag \\
\left\langle \left. \frac{\partial \mathcal{L}\left( \Psi \left( t\right)
\right) }{\partial \Psi }\right| \delta \Psi \left( t\right) \right\rangle
&=&\frac{1}{\left\langle \Psi \left( t\right) |\Psi \left( t\right)
\right\rangle }\left\{ \left\langle \Psi \left( t\right) \left| \left( i%
\frac{\partial }{\partial t}-H-\mathcal{L}\left( \Psi \left( t\right)
\right) I\right) \right. \delta \Psi \left( t\right) \right\rangle \right\}
\end{eqnarray}%
and also 
\begin{eqnarray}

\right\rangle &=&\int \delta \Psi \left( \mathsf{z,}t\right) ^{\ast }\frac{%

\begin{eqnarray}

}\left( \Psi \left( t\right) \right) I\right) \right. \mathsf{z}\right\rangle
\end{eqnarray}

\subsection{Equations in Terms of Lagrangian Density}

The Lagrangian is given in terms of its density is 
\begin{eqnarray}
\mathcal{L}\left( \Psi \left( t\right) \right) &=&\frac{1}{\left\langle \Psi
\left( t\right) |\Psi \left( t\right) \right\rangle }\int \Psi \left( 
\mathsf{z,}t\right) ^{\ast }\left\{ i\frac{\partial \Psi \left( \mathsf{z,}%
t\right) }{\partial t}-\left[ H\Psi \right] \left( \mathsf{z,}t\right)
\right\} d\mathsf{z} \\
&\equiv &\frac{1}{\left\langle \Psi \left( t\right) |\Psi \left( t\right)
\right\rangle }\int \Psi \left( \mathsf{z,}t\right) ^{\ast }\left\{ i\dot{%
\Psi}\left( \mathsf{z,}t\right) -\left[ H\Psi \right] \left( \mathsf{z,}%
t\right) \right\} d\mathsf{z}
\end{eqnarray}%
where 
\begin{equation}
\left[ H\Psi \right] \left( \mathsf{z,}t\right) =\int H\left( \mathsf{z,z}%
^{\prime }\right) \Psi \left( \mathsf{z}^{\prime }\mathsf{,}t\right) d%
\mathsf{z}^{\prime }
\end{equation}%
The Lagrangian needs to be considered explicitly depending on $\dot{\Psi}$,
i.e. $\mathcal{L=}$ $\mathcal{L}\left( \Psi \left( t\right) ,\dot{\Psi}%
\left( t\right) \right) $, in order to obtain the correct expressions for
the first order variation of the Action. This can be seen by observing that
the first order portion of the full variation 
\begin{equation}
\frac{\int \left[ \Psi +\delta \Psi \right] ^{\ast }\left( \mathsf{z,}%
t\right) \left\{ i\overset{\cdot }{\left[ \Psi +\delta \Psi \right] }\left( 
\mathsf{z,}t\right) -\left[ H\left[ \Psi +\delta \Psi \right] \right] \left( 
\mathsf{z,}t\right) \right\} d\mathsf{z}}{\left\langle \left[ \Psi +\delta
\Psi \right] \left( t\right) |\left[ \Psi +\delta \Psi \right] \left(
t\right) \right\rangle }
\end{equation}%
is given by 
\begin{eqnarray}
&&\delta \mathcal{L}\left( \Psi \left( t\right) \right) =  \notag \\
&&\frac{\int \delta \Psi ^{\ast }\left( \mathsf{z,}t\right) \left\{ i\dot{%
\Psi}\left( \mathsf{z,}t\right) -\left[ H\Psi \right] \left( \mathsf{z,}%

\left( t\right) \right\rangle }  \notag \\

\begin{equation}

\section{Adjoint of Time Displacement Operator\label{Appendix A}}

The adjoint of the time displacement operator can be expressed using 
\begin{equation}
i\frac{d}{dt}\left\langle \varphi _{1}\left( t\right) |\varphi _{2}\left(
t\right) \right\rangle =\left\langle -i\frac{\partial }{\partial t}\varphi
_{1}\left( t\right) |\varphi _{2}\left( t\right) \right\rangle +\left\langle
\varphi _{1}\left( t\right) |i\frac{\partial }{\partial t}\varphi _{2}\left(
t\right) \right\rangle \quad \forall \varphi _{1},\varphi _{2}
\end{equation}%
so that 
\begin{equation}
\left\langle -i\frac{\partial }{\partial t}\varphi _{1}\left( t\right)
|\varphi _{2}\left( t\right) \right\rangle =i\frac{d}{dt}\left\langle
\varphi _{1}\left( t\right) |\varphi _{2}\left( t\right) \right\rangle
-\left\langle \varphi _{1}\left( t\right) |i\frac{\partial }{\partial t}%
\varphi _{2}\left( t\right) \right\rangle \quad \forall \varphi _{1},\varphi
_{2},t
\end{equation}%
Noting that 
\begin{equation}
\left\langle \left[ i\frac{\partial }{\partial t}\right] ^{\dagger }\varphi
_{1}\left( t\right) |\varphi _{2}\left( t\right) \right\rangle =\left\langle
\varphi _{1}\left( t\right) |i\frac{\partial }{\partial t}\varphi _{2}\left(
t\right) \right\rangle \quad \forall \varphi _{1},\varphi _{2},t
\end{equation}%
one can conclude that 
\begin{equation}
\left\langle -i\frac{\partial }{\partial t}\varphi _{1}\left( t\right)
|\varphi _{2}\left( t\right) \right\rangle =i\frac{d}{dt}\left\langle
\varphi _{1}\left( t\right) |\varphi _{2}\left( t\right) \right\rangle
-\left\langle \left[ i\frac{\partial }{\partial t}\right] ^{\dagger }\varphi
_{1}\left( t\right) |\varphi _{2}\left( t\right) \right\rangle \quad \forall
\varphi _{1},\varphi _{2},t
\end{equation}%
giving 
\begin{equation}
\left\langle \left[ i\frac{\partial }{\partial t}\right] ^{\dagger }\varphi
_{1}\left( t\right) |\varphi _{2}\left( t\right) \right\rangle =i\frac{d}{dt}%
\left\langle \varphi _{1}\left( t\right) |\varphi _{2}\left( t\right)
\right\rangle +\left\langle i\frac{\partial }{\partial t}\varphi _{1}\left(
t\right) |\varphi _{2}\left( t\right) \right\rangle \quad \forall \varphi
_{1},\varphi _{2},t
\end{equation}%
This clearly shows that in general 
\begin{equation}
\left[ i\frac{\partial }{\partial t}\right] ^{\dagger }\neq i\frac{\partial 
}{\partial t}
\end{equation}%
unless 
\begin{equation}
\frac{d}{dt}\left\langle \varphi _{1}\left( t\right) |\varphi _{2}\left(
t\right) \right\rangle =0\quad \forall \varphi _{1},\varphi _{2},t
\end{equation}%
This clearly \emph{cannot} happen for arbitrary paths $\left\{ \varphi
_{1}\left( t\right) ,\varphi _{2}\left( t\right) \right\} $, but it can
occur along specific paths given by 
\begin{eqnarray}
\left| \varphi _{1}\left( t\right) \right\rangle &=&V_{1}\left( t\right)
\left| \varphi _{10}\right\rangle \\
\left| \varphi _{2}\left( t\right) \right\rangle &=&V_{2}\left( t\right)
\left| \varphi _{20}\right\rangle
\end{eqnarray}%
when%
\begin{equation}
\left\langle V_{1}\left( t\right) \varphi _{10}|V_{2}\left( t\right) \varphi
_{20}\right\rangle =\text{k}
\end{equation}%
where k is constant. This implies that 
\begin{equation}
\left\langle V_{1}\left( t\right) \varphi _{10}|V_{2}\left( t\right) \varphi
_{20}\right\rangle =\left\langle \varphi _{10}|V_{1}\left( t\right)
^{\dagger }V_{2}\left( t\right) \varphi _{20}\right\rangle =\left\langle
\varphi _{10}|\varphi _{20}\right\rangle \quad \forall t
\end{equation}%
If we set 
\begin{equation}
V_{1}\left( 0\right) =V_{2}\left( 0\right) =I
\end{equation}%
If we want $\left\langle \varphi _{1}\left( t\right) |\varphi _{2}\left(
t\right) \right\rangle $ to be constant for arbitrary initial states $%
\left\{ \left| \varphi _{10}\right\rangle ,\left| \varphi _{20}\right\rangle
\right\} $ then 
\begin{equation}
V_{1}\left( t\right) =V_{2}\left( t\right) =U\left( t\right)
\end{equation}%
where 
\begin{equation}
U\left( t\right) ^{\dagger }U\left( t\right) =U\left( t\right) U\left(
t\right) ^{\dagger }=I
\end{equation}%
Note that even in this case $i\frac{\partial }{\partial t}$ is \emph{not }%
self adjoint, it is only self adjoint when restricted to the paths $U\left(
t\right) \left| \varphi \right\rangle $ for arbitrary, but fixed initial
states $\left| \varphi _{10}\right\rangle ,\left| \varphi _{20}\right\rangle 
$.

We can also use this analysis to relate derivatives of the paths $\left\{
\varphi _{1}\left( t\right) ,\varphi _{2}\left( t\right) \right\} $ as 
\begin{equation}
\left\langle \frac{\partial }{\partial t}\varphi _{1}\left( t\right)
|\varphi _{2}\left( t\right) \right\rangle =\frac{d}{dt}\left\langle \varphi
_{1}\left( t\right) |\varphi _{2}\left( t\right) \right\rangle -\left\langle
\varphi _{1}\left( t\right) |\frac{\partial }{\partial t}\varphi _{2}\left(
t\right) \right\rangle \quad \forall \varphi _{1},\varphi _{2},t
\end{equation}%
giving 
\begin{eqnarray}
\left\langle \dot{\varphi}_{1}\left( t\right) |\varphi _{2}\left( t\right)
\right\rangle &=&\frac{d}{dt}\left\langle \varphi _{1}\left( t\right)
|\varphi _{2}\left( t\right) \right\rangle -\left\langle \varphi _{1}\left(
t\right) |\dot{\varphi}_{2}\left( t\right) \right\rangle \\
\left\langle \varphi _{1}\left( t\right) |\dot{\varphi}_{2}\left( t\right)
\right\rangle &=&\frac{d}{dt}\left\langle \varphi _{1}\left( t\right)
|\varphi _{2}\left( t\right) \right\rangle -\left\langle \dot{\varphi}%
_{1}\left( t\right) |\varphi _{2}\left( t\right) \right\rangle
\end{eqnarray}%
so in particular 
\begin{equation}
\left\langle \delta \dot{\varphi}\left( t\right) |\varphi \left( t\right)
\right\rangle =\frac{d}{dt}\left\langle \delta \varphi \left( t\right)
|\varphi \left( t\right) \right\rangle -\left\langle \delta \varphi \left(
t\right) |\dot{\varphi}\left( t\right) \right\rangle
\end{equation}
