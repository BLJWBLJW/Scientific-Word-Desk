\section{Vectors as Matrices}%

\begin{equation}
\mathbf{v}=\left(
\begin{array}
[c]{c}%
v_{1}\\
v_{2}%
\end{array}
\right)  \in\mathbb{C}^{2}\hookrightarrow\left(
\begin{array}
[c]{cc}%
v_{1} & 0\\
v_{2} & 0
\end{array}
\right)  =\mathbb{V}_{v}\in\mathcal{B}\left(  \mathbb{C}^{2}\right)
\end{equation}%
\begin{equation}
\mathbb{A}\mathbf{v}=\mathbb{A}\left(
\begin{array}
[c]{c}%
v_{1}\\
v_{2}%
\end{array}
\right)  \in\mathbb{C}^{2}\hookrightarrow\mathbb{A}\left(
\begin{array}
[c]{cc}%
v_{1} & 0\\
v_{2} & 0
\end{array}
\right)  =\left(
\begin{array}
[c]{cc}%
\mathbb{A}\mathbf{v} & 0
\end{array}
\right)  =\mathbb{AV}_{v}\in\mathcal{B}\left(  \mathbb{C}^{2}\right)
\end{equation}%
\begin{equation}
\mathbb{V}_{v}\mathbb{A=}\left(
\begin{array}
[c]{cc}%
v_{1} & 0\\
v_{2} & 0
\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
a_{11} & a_{12}\\
a_{21} & a_{21}%
\end{array}
\right)  =\left(
\begin{array}
[c]{cc}%
v_{1}a_{11} & v_{1}a_{12}\\
v_{2}a_{11} & v_{2}a_{12}%
\end{array}
\right)  \notin\mathcal{B}\left(  \mathbb{C}^{2}\right)
\end{equation}%
\begin{equation}
\mathbb{V}_{v}\mathbb{A}^{\dagger}=\left(
\begin{array}
[c]{cc}%
v_{1} & 0\\
v_{2} & 0
\end{array}
\right)  \overline{\left(
\begin{array}
[c]{cc}%
a_{11} & a_{21}\\
a_{12} & a_{22}%
\end{array}
\right)  }=\left(
\begin{array}
[c]{cc}%
v_{1}\overline{a_{11}} & v_{1}\overline{a_{21}}\\
v_{2}\overline{a_{11}} & v_{2}\overline{a_{21}}%
\end{array}
\right)
\end{equation}%
\begin{equation}
\mathbb{V}_{v}^{\dagger}\mathbb{A}^{\dagger}=\overline{\left(
\begin{array}
[c]{cc}%
v_{1} & v_{2}\\
0 & 0
\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
a_{11} & a_{21}\\
a_{12} & a_{22}%
\end{array}
\right)  }=\left(
\begin{array}
[c]{cc}%
\overline{v_{1}a_{11}+v_{2}a_{12}} & \overline{v_{1}a_{21}+v_{2}a_{22}}\\
0 & 0
\end{array}
\right)
\end{equation}%
\begin{equation}
\mathbb{U}\left(  t\right)  \mathbb{V}_{v}\mathbb{V}_{v}^{\dagger}%
\mathbb{U}\left(  t\right)  ^{\dagger}=\mathbb{U}\left(  t\right)
\mathbb{V}_{v}\left(  \mathbb{U}\left(  t\right)  \mathbb{V}_{v}\right)
^{\dagger}=\mathbb{U}\left(  t\right)  \mathbb{V}_{v}\mathbb{U}\left(
t\right)  ^{\dagger}\mathbb{U}\left(  t\right)  \mathbb{V}_{v}^{\dagger
}\mathbb{U}\left(  t\right)  ^{\dagger}%
\end{equation}%
\begin{equation}
\mathbb{U}\left(  t\right)  \mathbb{V}_{v}=\exp\left\{  -i\mathbb{H}t\right\}
\mathbb{V}_{v}=e^{-Et}\mathbb{V}_{v}%
\end{equation}%
\begin{equation}
\left(  \mathbb{U}\left(  t\right)  \mathbb{V}_{v}\right)  ^{\dagger}=\left(
\exp\left\{  -i\mathbb{H}t\right\}  \mathbb{V}_{v}\right)  ^{\dagger}%
=e^{Et}\mathbb{V}_{v}^{\dagger}%
\end{equation}%
\begin{equation}
\mathbb{U}\left(  t\right)  \mathbb{V}_{v}\mathbb{V}_{v}^{\dagger}%
\mathbb{U}\left(  t\right)  ^{\dagger}=\mathbb{U}\left(  t\right)
\mathbb{V}_{v}\left(  \mathbb{U}\left(  t\right)  \mathbb{V}_{v}\right)
^{\dagger}=\mathbb{V}_{v}\mathbb{V}_{v}^{\dagger}%
\end{equation}%
\begin{equation}
-i\frac{d}{dt}\left\{  \left(  \mathbb{V}_{v}\mathbb{V}_{v}^{\dagger}\right)
\left(  t\right)  \right\}  =\left[  \mathbb{H},\left(  \mathbb{V}%
_{v}\mathbb{V}_{v}^{\dagger}\right)  \left(  t\right)  \right]
\end{equation}
compare
\begin{equation}
\mathbb{U}\left(  t\right)  \mathbb{V}_{v}=\left\{  I-i\mathbb{H}t-\frac{1}%
{2}\mathbb{H}^{2}t^{2}+\cdots\right\}  \mathbb{V}_{v}\in\mathcal{B}\left(
\mathbb{C}^{2}\right)
\end{equation}
with
\begin{equation}
\mathbb{U}\left(  t\right)  \mathbb{V}_{v}\mathbb{U}\left(  t\right)
^{\dagger}=\left\{  I-i\mathbb{H}t-\frac{1}{2}\mathbb{H}^{2}t^{2}%
+\cdots\right\}  \mathbb{V}_{v}\left\{  I+i\mathbb{H}t-\frac{1}{2}%
\mathbb{H}^{2}t^{2}+\cdots\right\}  \notin\mathcal{B}\left(  \mathbb{C}%
^{2}\right)
\end{equation}

\begin{equation}
-i\frac{d}{dt}\left\{  \mathbb{V}_{v}\left(  t\right)  \right\}  =\left[
\mathbb{H},\mathbb{V}_{v}\left(  t\right)  \right]  \notin\mathcal{B}\left(
\mathbb{C}^{2}\right)
\end{equation}%
\begin{equation}
\mathbf{v}_{1}^{\dagger}\mathbf{v}_{2\hookrightarrow}\operatorname{Tr}\left\{
\mathbb{V}_{v1}^{\dagger}\mathbb{V}_{v2}\right\}  =\operatorname{Tr}\left\{
\left(
\begin{array}
[c]{cc}%
\mathbf{v}_{1}^{\dagger}\mathbf{v}_{2} & 0\\
0 & 0
\end{array}
\right)  \right\}
\end{equation}%
\begin{equation}
\mathbb{V}_{v1}\mathbb{V}_{v2}=\left(
\begin{array}
[c]{cc}%
v_{11} & 0\\
v_{12} & 0
\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
v_{21} & 0\\
v_{22} & 0
\end{array}
\right)  =\left(
\begin{array}
[c]{cc}%
v_{11}v_{21} & 0\\
v_{12}v_{21} & 0
\end{array}
\right)
\end{equation}%
