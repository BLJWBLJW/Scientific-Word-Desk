\section{Integral Matrix Elements}

These matrices have the following symmetries and are defined in the
following manner: 
\begin{eqnarray}
&&h_{jk}^{a}=h_{j+sk+s}^{a}=-\int \chi _{j}^{\ast }\left( \mathbf{r}\right)
\left\{ \sum_{_{1\leq j\leq N_{nuc}}}\frac{Z_{j}}{\left\| \mathbf{R}_{j}-%
\mathbf{r}\right\| }+\frac{1}{2m_{e}}\nabla _{\mathbf{r}}^{2}\right\} \chi
_{k}\left( \mathbf{r}\right) d\mathbf{r};\quad 1\leq j,k\leq s  \notag \\
&&h_{jk+s}^{a}=h_{j+sk}^{a};\quad 1\leq j,k\leq s \\
&&  \notag \\
&&\left\langle jk||lm\right\rangle ^{a}=\int \left\{ \chi _{j}^{\ast }\left( 
\mathbf{r}_{1}\right) \chi _{k}^{\ast }\left( \mathbf{r}_{2}\right) \frac{1}{%
\left\| \mathbf{r}_{1}-\mathbf{r}_{2}\right\| }\chi _{l}\left( \mathbf{r}%
_{1}\right) \chi _{m}\left( \mathbf{r}_{2}\right) \right.   \notag \\
&&\qquad -\left. \chi _{j}^{\ast }\left( \mathbf{r}_{1}\right) \chi
_{k}^{\ast }\left( \mathbf{r}_{2}\right) \frac{1}{\left\| \mathbf{r}_{1}-%
\mathbf{r}_{2}\right\| }\chi _{l}\left( \mathbf{r}_{2}\right) \chi
_{m}\left( \mathbf{r}_{1}\right) \right\} d\mathbf{r}_{1}d\mathbf{r}%
_{2};\quad 1\leq j,k\leq s;1\leq l,m\leq s \\
&&  \notag \\
&&\left\langle j+sk||lm\right\rangle ^{a}=\left\langle jk+s||lm\right\rangle
^{a}=\left\langle jk||l+sm\right\rangle ^{a}=\left\langle
jk||lm+s\right\rangle ^{a}=\left\langle j+sk+s||l+sm\right\rangle ^{a}= 
\notag \\
&&\left\langle j+sk+s||lm+s\right\rangle ^{a}=\left\langle
j+sk||l+sm+s\right\rangle ^{a}=\left\langle jk+s||l+sm+s\right\rangle ^{a}=0
\notag \\
&&\left. {}\right. \quad \qquad \qquad \quad \qquad \qquad \quad \qquad
\qquad \quad \qquad \qquad \quad 1\leq j,k\leq s;1\leq l,m\leq s \\
&&  \notag \\
&&\left\langle jk+s||l+sm\right\rangle ^{a}=\left\langle
j+sk||l+sm\right\rangle ^{a}=  \notag \\
&&\int \chi _{j}^{\ast }\left( \mathbf{r}_{1}\right) \chi _{k}^{\ast }\left( 
\mathbf{r}_{2}\right) \frac{1}{\left\| \mathbf{r}_{1}-\mathbf{r}_{2}\right\| 
}\chi _{l}\left( \mathbf{r}_{1}\right) \chi _{m}\left( \mathbf{r}_{2}\right)
d\mathbf{r}_{1}d\mathbf{r}_{2};\quad 1\leq j,k\leq s;1\leq l,m\leq s \\
&&  \notag \\
&&\left\langle jk+s||lm+s\right\rangle ^{a}=\left\langle
j+sk||l+sm\right\rangle ^{a}=-\left\langle jk+s||l+sm\right\rangle
^{a}=-\left\langle j+sk||lm+s\right\rangle ^{a}  \notag \\
&&\left. {}\right. \quad \qquad \qquad \quad \qquad \qquad \quad \quad
\qquad \quad \qquad \qquad \qquad \quad 1\leq j,k\leq s;1\leq l,m\leq s \\
&&\left\langle jk|||jk\right\rangle ^{a}=\left\langle jk||jk\right\rangle
^{a}+\left\langle j\,k+s||j\,k+s\right\rangle ^{a}+\left\langle
j+s\,k||j+s\,k\right\rangle ^{a}+\left\langle
j+s\,k+s||j+s\,k+s\right\rangle ^{a}  \notag \\
&&\left. {}\right. \quad \qquad \qquad \quad \qquad \qquad =0;\quad \qquad
\qquad \quad \qquad \qquad 1\leq j,k\leq s;1\leq l,m\leq s
\end{eqnarray}%
We concentrate on $\left\langle jk||lm\right\rangle ^{a}$ 
\begin{eqnarray}
&&\left\langle jk||lm\right\rangle ^{a}=\int \left\{ \chi _{j}^{\ast }\left( 
\mathbf{r}_{1}\right) \chi _{k}^{\ast }\left( \mathbf{r}_{2}\right) \frac{1}{%
\left\| \mathbf{r}_{1}-\mathbf{r}_{2}\right\| }\chi _{l}\left( \mathbf{r}%
_{1}\right) \chi _{m}\left( \mathbf{r}_{2}\right) \right.   \notag \\
&&\qquad -\left. \chi _{j}^{\ast }\left( \mathbf{r}_{1}\right) \chi
_{k}^{\ast }\left( \mathbf{r}_{2}\right) \frac{1}{\left\| \mathbf{r}_{1}-%
\mathbf{r}_{2}\right\| }\chi _{l}\left( \mathbf{r}_{2}\right) \chi
_{m}\left( \mathbf{r}_{1}\right) \right\} d\mathbf{r}_{1}d\mathbf{r}%
_{2};\quad 1\leq j,k\leq s;1\leq l,m\leq s
\end{eqnarray}%
We can define 
\begin{equation}
J_{jklm}=\int \chi _{j}^{\ast }\left( \mathbf{r}_{1}\right) \chi _{k}^{\ast
}\left( \mathbf{r}_{2}\right) \frac{1}{\left\| \mathbf{r}_{1}-\mathbf{r}%
_{2}\right\| }\chi _{l}\left( \mathbf{r}_{1}\right) \chi _{m}\left( \mathbf{r%
}_{2}\right) d\mathbf{r}_{1}d\mathbf{r}_{2}
\end{equation}%
then 
\begin{equation}
\langle jk||lm\rangle ^{a}=J_{jklm}-J_{jkml}
\end{equation}%
We also have the symmetry 
\begin{equation}
J_{jklm}=J_{kjml}
\end{equation}%
The matrix $\mathbb{J}$ is positive definite as 
\begin{eqnarray}
\mathbf{v}^{\dagger }\mathbb{J}\mathbf{v} &\mathbf{=}&\sum_{1\leq
j,k,l,m\leq s}v_{jk}^{\ast }\int \chi _{j}^{\ast }\left( \mathbf{r}%
_{1}\right) \chi _{k}^{\ast }\left( \mathbf{r}_{2}\right) \frac{1}{\left\| 
\mathbf{r}_{1}-\mathbf{r}_{2}\right\| }\chi _{l}\left( \mathbf{r}_{1}\right)
\chi _{m}\left( \mathbf{r}_{2}\right) d\mathbf{r}_{1}d\mathbf{r}_{2}v_{lm} 
\notag \\
&=&\int \Phi \left( \mathbf{r}_{1},\mathbf{r}_{2}\right) ^{\ast }\frac{1}{%
\left\| \mathbf{r}_{1}-\mathbf{r}_{2}\right\| }\Phi \left( \mathbf{r}_{1},%
\mathbf{r}_{2}\right) d\mathbf{r}_{1}d\mathbf{r}_{2}\geq 0
\end{eqnarray}%
Thus it is hermitian (and if the atomic orbital basis is real it real
symmetric). Hence 
\begin{equation}
\mathbb{J=Q}^{\dagger }\mathbb{Q=U}^{\dagger }\mathbf{\eta }\mathbb{U}
\end{equation}%
where $\mathbf{\eta }$ are the eigenvalues of $\mathbb{J}$. Thus 
\begin{equation}
\left\langle jk||lm\right\rangle ^{a}=\left( \mathbb{Q}^{\dagger }\mathbb{Q}%
\right) _{jklm}-\left( \mathbb{Q}^{\dagger }\mathbb{Q}\right)
_{jkml}=\sum_{1\leq a,b\leq s}Q_{jkab}^{\dagger }\left\{
Q_{ablm}-Q_{abml}\right\} =\sum_{1\leq a,b\leq s}Q_{jkab}^{\dagger }P_{ablm}
\end{equation}%
and 
\begin{equation}
\left\langle jk||lm\right\rangle ^{a}=\left( \mathbb{U}^{\dagger }\mathbf{%
\eta }\mathbb{U}\right) _{jklm}-\left( \mathbb{U}^{\dagger }\mathbf{\eta }%
\mathbb{U}\right) _{jkml}=\sum_{1\leq a,b\leq s}U_{jkab}^{\dagger }\eta
_{ab}\left\{ U_{ablm}-U_{abml}\right\} =\sum_{1\leq a,b\leq s}\mathcal{R}%
_{jkab}^{\dagger }\mathcal{S}_{ablm}
\end{equation}%
where 
\begin{equation}
P_{ablm}=Q_{ablm}-Q_{abml}
\end{equation}%
and 
\begin{eqnarray}
\mathcal{S}_{ablm} &=&U_{ablm}-U_{abml}  \notag \\
R_{ablm}^{\dagger } &=&U_{jkab}^{\dagger }\eta _{ab}
\end{eqnarray}%
Approximation schemes can be based on ignoring $\eta _{ab}\leq cutoff$.

^{r}\left\{ V_{f\,i_{1}p}V_{f\,j_{2}q}-V_{f\,i_{1}q}V_{f\,i_{2}p}\right\}
