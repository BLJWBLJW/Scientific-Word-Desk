\section{Poisson Brackets and Symbols}

The Poisson Brackets $\left\{  f,g\right\}  $ between two complex valued
functions $\left(  f,g\right)  $ on the curved phase space $\mathcal{M}$ is a
complex valued bilinear map from the tangent bundle $\mathfrak{T}$ of
$\mathcal{M}$
\begin{equation}
\left\{  ,\right\}  :\mathfrak{T}\times\mathfrak{T}\rightarrow\mathbb{C},
\end{equation}
defined on \emph{each} tangent space as
\begin{align}
\left\{  f\left(  \mathbf{z}\right)  ,g\left(  \mathbf{z}\right)  \right\}
&  =\omega\left(  \mathbf{z}\right)  ^{-1}\left(  X_{f}\left(  \mathbf{z}%
\right)  ,X_{g}\left(  \mathbf{z}\right)  \right) \nonumber\\
&  =\left(
\begin{array}
[c]{cc}%
\frac{\partial f\left(  \mathbf{z}\right)  }{\partial\mathbf{\bar{z}}} &
\frac{\partial f\left(  \mathbf{z}\right)  }{\partial\mathbf{z}}%
\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
\mathbf{0} & i\overline{\mathbf{C}\left(  \mathbf{\bar{z},z}\right)  }^{-1}\\
-i\mathbf{C}\left(  \mathbf{\bar{z},z}\right)  ^{-1} & \mathbf{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\frac{\partial g\left(  \mathbf{z}\right)  }{\partial\mathbf{\bar{z}}}\\
\frac{\partial g\left(  \mathbf{z}\right)  }{\partial\mathbf{z}}%
\end{array}
\right)  . \label{PB}%
\end{align}
which in detail is
\begin{equation}
i\sum_{1\leq j,k\leq\nu}\left\{  \frac{\partial f\left(  \mathbf{z}\right)
}{\partial\bar{z}_{j}}\overline{C_{jk}\left(  \mathbf{\bar{z},z}\right)
}^{-1}\frac{\partial g\left(  \mathbf{z}\right)  }{\partial z_{k}}%
-\frac{\partial f\left(  \mathbf{z}\right)  }{\partial z_{j}}C_{jk}\left(
\mathbf{\bar{z},z}\right)  ^{-1}\frac{\partial g\left(  \mathbf{z}\right)
}{\partial\bar{z}_{k}}\right\}  .
\end{equation}
Real valued classical observables $\mathcal{O}\left(  \mathbf{\bar{z}%
,z}\right)  $ that are the symbols of quantum observables are defined in
general by
\begin{equation}
\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  =\frac{\left\langle \left.
\Phi\left(  \mathbf{z}\right)  \right\vert O\Phi\left(  \mathbf{z}\right)
\right\rangle }{\left\langle \left.  \Phi\left(  \mathbf{z}\right)
\right\vert \Phi\left(  \mathbf{z}\right)  \right\rangle }=\frac
{\mathcal{O}_{u}\left(  \mathbf{\bar{z},z}\right)  }{\mathcal{S}\left(
\mathbf{\bar{z},z}\right)  }.
\end{equation}
In the special case of $SU\left(  \mathcal{H}^{N}\right)  /U\left(  1\right)
$-CS's these symbols and the metric matrix become%
\begin{equation}
\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  =\frac{\mathbf{z}^{\dagger
}\mathbb{O}\mathbf{z}}{\mathbf{z}^{\dagger}\mathbf{z}}%
\end{equation}
and
\begin{equation}
\left(
\begin{array}
[c]{cc}%
\mathbf{0} & i\overline{\mathbf{C}\left(  \mathbf{\bar{z},z}\right)  }^{-1}\\
-i\mathbf{C}\left(  \mathbf{\bar{z},z}\right)  ^{-1} & \mathbf{0}%
\end{array}
\right)  =\left(
\begin{array}
[c]{cc}%
\mathbf{0} & i\mathbf{P}_{\mathbf{\bar{z}}}^{\bot}\\
-i\mathbf{P}_{\mathbf{z}}^{\bot} & \mathbf{0}%
\end{array}
\right)  ,
\end{equation}
(taking the generalized inverse).

The 1$^{\text{st }}$Derivatives of the symbols are
\begin{equation}
\frac{\partial\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  }{\partial
\mathbf{x}}=\frac{1}{\mathcal{S}\left(  \mathbf{\bar{z},z}\right)  }\left\{
\frac{\partial\mathcal{O}_{u}\left(  \mathbf{\bar{z},z}\right)  }%
{\partial\mathbf{x}}-\mathcal{O}\left(  \mathbf{\bar{z},z}\right)
\frac{\partial\mathcal{S}\left(  \mathbf{\bar{z},z}\right)  }{\partial
\mathbf{x}}\right\}  ,
\end{equation}
which for $SU\left(  \mathcal{H}^{N}\right)  /U\left(  1\right)  $-CS's gives%
\begin{align}
D^{1}\mathcal{O}\left(  \mathbf{\bar{z},z}\right)   &  =\left(
\begin{array}
[c]{c}%
\frac{\partial\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  }{\partial
\mathbf{\bar{z}}}\\
\frac{\partial\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  }{\partial
\mathbf{z}}%
\end{array}
\right)  =\frac{1}{\mathbf{z}^{\dagger}\mathbf{z}}\left(
\begin{array}
[c]{c}%
\left\{  \mathbb{O}-\mathcal{O}\left(  \mathbf{\bar{z},z}\right)
\mathbb{I}\right\}  \mathbf{z}\\
\left\{  \overline{\mathbb{O}}-\mathcal{O}\left(  \mathbf{\bar{z},z}\right)
\mathbb{I}\right\}  \mathbf{\bar{z}}%
\end{array}
\right) \\
&  =\left(
\begin{array}
[c]{cc}%
\mathbb{M}_{\mathcal{O}} & \mathbf{0}\\
\mathbf{0} & \overline{\mathbb{M}}_{\mathcal{O}}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\mathbf{z}\\
\mathbf{\bar{z}}%
\end{array}
\right)  =\left(
\begin{array}
[c]{cc}%
\mathbf{0} & \mathbb{M}_{\mathcal{O}}\\
\overline{\mathbb{M}}_{\mathcal{O}} & \mathbf{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\mathbf{\bar{z}}\\
\mathbf{z}%
\end{array}
\right)  .
\end{align}
For $SU\left(  \mathcal{H}^{N}\right)  /U\left(  1\right)  $-CS's the vector
$\frac{\partial\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  }{\partial
\mathbf{\bar{z}}}$ is perpendicular to $\mathbf{z}$ and $\frac{\partial
\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  }{\partial\mathbf{z}}$ is
perpendicular to $\mathbf{\bar{z}}$ as shown by
\begin{align}
\mathbf{z}^{\dagger}\frac{\partial\mathcal{O}\left(  \mathbf{\bar{z}%
,z}\right)  }{\partial\mathbf{\bar{z}}}  &  =\mathbf{z}^{\dagger}\left\{
\mathbb{O}-\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  \mathbb{I}\right\}
\mathbf{z=z}^{\dagger}\mathbb{O}\mathbf{z-}\mathcal{O}\left(  \mathbf{\bar
{z},z}\right)  \mathbf{z}^{\dagger}\mathbf{z=}\mathcal{O}\left(
\mathbf{\bar{z},z}\right)  \mathbf{z}^{\dagger}\mathbf{z-}\mathcal{O}\left(
\mathbf{\bar{z},z}\right)  \mathbf{z}^{\dagger}\mathbf{z=0}\nonumber\\
\mathbf{z}^{t}\frac{\partial\mathcal{O}\left(  \mathbf{\bar{z},z}\right)
}{\partial\mathbf{z}}  &  =\mathbf{z}^{t}\left\{  \overline{\mathbb{O}%
}-\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  \mathbb{I}\right\}
\mathbf{\bar{z}=}\overline{\mathbf{z}^{\dagger}\left\{  \mathbb{O}%
-\mathcal{O}\left(  \mathbf{\bar{z},z}\right)  \mathbb{I}\right\}  \mathbf{z}%
}=0.
\end{align}

\right)  \left(
\begin{array}
[c]{c}%
\delta\mathbf{\bar{z}}\\
\delta\mathbf{z}%
\end{array}
\right)  =\nonumber\\
&  \frac{1}{2}\left\{  \left(

_{0}\right)  }{\partial\mathbf{\bar{z}}\partial\mathbf{\bar{z}}} &
\frac{\partial^{2}\mathcal{O}\left(  \mathbf{\bar{z}}_{0}\mathbf{,z}%
_{0}\right)  }{\partial\mathbf{\bar{z}}\partial\mathbf{z}}\\
\frac{\partial^{2}\mathcal{O}\left(  \mathbf{\bar{z}}_{0}\mathbf{,z}%
_{0}\right)  }{\partial\mathbf{z}\partial\mathbf{\bar{z}}} & \frac
{\partial^{2}\mathcal{O}\left(  \mathbf{\bar{z}}_{0}\mathbf{,z}_{0}\right)

\right)  \left(
\begin{array}

\right)  +\left(
\begin{array}

\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
\frac{\partial^{2}\mathcal{O}\left(  \mathbf{\bar{z}}_{0}\mathbf{,z}%
_{0}\right)  }{\partial\mathbf{\bar{z}}\partial\mathbf{\bar{z}}} &
\frac{\partial^{2}\mathcal{O}\left(  \mathbf{\bar{z}}_{0}\mathbf{,z}%

\frac{\partial^{2}\mathcal{O}\left(  \mathbf{\bar{z}}_{0}\mathbf{,z}%

In the $SU\left(  \mathcal{H}^{N}\right)  /U\left(  1\right)  $-CS case this

\mathcal{O}_{2}\left[  \mathbf{\bar{z}}_{0}\mathbf{,z}_{0}\right]  \left(

\subsection{Time Evolution}

One can evaluate the PB\ along time dependent paths $\mathrm{z}\left(
t\right)  $ by
\begin{align}
\left\{  f\left(  \mathrm{z}\left(  t\right)  \right)  ,g\left(
\mathrm{z}\left(  t\right)  \right)  \right\}   &  =\omega\left(
\mathrm{z}\left(  t\right)  \right)  ^{-1}\left(  X_{f}\left(  \mathrm{z}%
\left(  t\right)  \right)  ,X_{g}\left(  \mathrm{z}\left(  t\right)  \right)
\right) \nonumber\\
&  =\left(
\begin{array}
[c]{cc}%
\frac{\partial f\left(  \mathrm{z}\left(  t\right)  \right)  }{\partial
\mathbf{\bar{z}}} & \frac{\partial f\left(  \mathrm{z}\left(  t\right)
\right)  }{\partial\mathbf{z}}%
\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
\mathbf{0} & i\overline{\mathbf{C}\left(  \mathrm{\bar{z}}\left(  t\right)
\mathbf{,}\mathrm{z}\left(  t\right)  \right)  }^{-1}\\
-i\mathbf{C}\left(  \mathrm{\bar{z}}\left(  t\right)  \mathbf{,}%
\mathrm{z}\left(  t\right)  \right)  ^{-1} & \mathbf{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\frac{\partial g\left(  \mathrm{z}\left(  t\right)  \right)  }{\partial
\mathbf{\bar{z}}}\\
\frac{\partial g\left(  \mathrm{z}\left(  t\right)  \right)  }{\partial
\mathbf{z}}%
\end{array}
\right)  ,
\end{align}
where all the derivatives are evaluated at the point $\mathrm{z}\left(
t\right)  .$

In the case of $SU\left(  \mathcal{H}^{N}\right)  /U\left(  1\right)  $-CS's,
the solution paths are determined from the classical EOM\ and phase equation
as
\begin{equation}
\mathrm{z}\left(  t\right)  =e^{-i\mathbb{H}t}\mathbf{z}_{i}%
\end{equation}
(note that in general $\mathbf{z}_{i}$ are not stationary points) and the time
dependent CS symbols are given by
\begin{equation}
\mathcal{O}\left(  \mathrm{\bar{z}}\left(  t\right)  \mathrm{,z}\left(
t\right)  \right)  =\frac{\mathrm{z}\left(  t\right)  ^{\dagger}%
\mathbb{O}\mathrm{z}\left(  t\right)  }{\mathrm{z}\left(  t\right)  ^{\dagger
}\mathrm{z}\left(  t\right)  }=\frac{\left[  e^{-i\mathbb{H}t}\mathbf{z}%
_{i}\right]  ^{\dagger}\mathbb{O}\left[  e^{-i\mathbb{H}t}\mathbf{z}%
_{i}\right]  }{\left[  e^{-i\mathbb{H}t}\mathbf{z}_{i}\right]  ^{\dagger
}\left[  e^{-i\mathbb{H}t}\mathbf{z}_{i}\right]  }=\frac{\mathbf{z}%
_{i}^{\dagger}\left(  e^{i\mathbb{H}t}\mathbb{O}e^{-i\mathbb{H}t}\right)
\mathbf{z}_{i}}{\mathbf{z}_{i}^{\dagger}\mathbf{z}_{i}}=\frac{\mathbf{z}%
_{i}^{\dagger}\mathbb{O}\left(  t\right)  \mathbf{z}_{i}}{\mathbf{z}%
_{i}^{\dagger}\mathbf{z}_{i}}.
\end{equation}
In this case the action of the Poisson Bracket is defined in terms of
derivatives \emph{with respect to the initial point}.\emph{ }We can extended
the PB to \emph{different} times by using a time evolution that maps the
tangent space $\mathcal{T}_{\mathrm{z}\left(  t\right)  }\rightarrow
\mathcal{T}_{\mathrm{z}\left(  t\right)  }$ and produces paths $\mathrm{z}%
_{L}\left(  t\right)  $ in $\mathcal{T}_{\mathrm{z}\left(  t\right)  }$ to
give
\begin{align}
&  \left\{  f\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}%
_{L}\left(  t_{1}\right)  \right)  ,g\left(  \mathrm{\bar{z}}_{L}\left(
t_{2}\right)  \mathrm{,z}_{L}\left(  t_{2}\right)  \right)  \right\}
=\omega_{\left(  \mathrm{\bar{z}}\left(  t\right)  \mathrm{,z}\left(
t\right)  \right)  }{}^{-1}\left(  X_{f}\left(  t_{1}\right)  ,X_{g}\left(
t_{2}\right)  \right) \nonumber\\
&  =\left(
\begin{array}
[c]{cc}%
\frac{\partial f\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)
\mathrm{,z}_{L}\left(  t_{1}\right)  \right)  }{\partial\mathbf{\bar{z}}_{i}}
& \frac{\partial f\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)
\mathrm{,z}_{L}\left(  t_{1}\right)  \right)  }{\partial\mathbf{z}_{i}}%
\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
\mathbf{0} & i\mathbf{P}_{\mathbf{\bar{z}}_{i}}^{\bot}\\
-i\mathbf{P}_{\mathbf{z}_{i}}^{\bot} & \mathbf{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\frac{\partial g\left(  \mathrm{\bar{z}}_{L}\left(  t_{2}\right)
\mathrm{,z}_{L}\left(  t_{2}\right)  \right)  }{\partial\mathbf{\bar{z}}_{i}%
}\\
\frac{\partial g\left(  \mathrm{\bar{z}}_{L}\left(  t_{2}\right)
\mathrm{,z}_{L}\left(  t_{2}\right)  \right)  }{\partial\mathbf{z}_{i}}%
\end{array}
\right)  ,
\end{align}
which gives for $SU\left(  \mathcal{H}^{N}\right)  /U\left(  1\right)  $-CS's%
\begin{align}
&  \left\{  f\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}%
_{L}\left(  t_{1}\right)  \right)  ,g\left(  \mathrm{\bar{z}}_{L}\left(
t_{2}\right)  \mathrm{,z}_{L}\left(  t_{2}\right)  \right)  \right\}
_{\mathrm{z}\left(  t\right)  }=\nonumber\\
&  \qquad\frac{1}{\left(  \mathbf{z}_{i}^{\dagger}\mathbf{z}_{i}\right)  ^{3}%
}\left(
\begin{array}
[c]{cc}%
\mathbf{\bar{z}}_{i}^{\dagger}\left\{  \mathbb{\bar{F}}\left(  t_{1}\right)
-\mathcal{F}\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}%
_{L}\left(  t_{1}\right)  \right)  \mathbb{I}\right\}  & \mathbf{z}%
_{i}^{\dagger}\left\{  \mathbb{F}\left(  t_{1}\right)  -\mathcal{F}\left(
\mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}_{L}\left(  t_{1}\right)
\right)  \mathbb{I}\right\}
\end{array}
\right) \nonumber\\
&  \qquad\qquad\qquad\left(
\begin{array}
[c]{cc}%
\mathbf{0} & i\mathbf{P}_{\mathbf{\bar{z}}_{i}}^{\bot}\\
-i\mathbf{P}_{\mathbf{z}_{i}}^{\bot} & \mathbf{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\left\{  \mathbb{G}\left(  t_{2}\right)  -\mathcal{G}\left(  \mathrm{\bar{z}%
}_{L}\left(  t_{2}\right)  \mathrm{,z}_{L}\left(  t_{2}\right)  \right)
\mathbb{I}\right\}  \mathbf{z}_{i}\\
\left\{  \mathbb{\bar{G}}\left(  t_{2}\right)  -\mathcal{G}\left(
\mathrm{\bar{z}}_{L}\left(  t_{2}\right)  \mathrm{,z}_{L}\left(  t_{2}\right)
\right)  \mathbb{I}\right\}  \mathbf{\bar{z}}_{i}%
\end{array}
\right)  ,
\end{align}
producing
\begin{align}
&  \left\{  f\left(  t_{1}\right)  ,g\left(  t_{2}\right)  \right\}
=\quad\frac{1}{\left(  \mathbf{z}_{i}^{\dagger}\mathbf{z}_{i}\right)  ^{3}%
}\left(
\begin{array}
[c]{cc}%
\mathbf{\bar{z}}_{i}^{\dagger}\left\{  \mathbb{\bar{F}}\left(  t_{1}\right)
-\mathcal{F}\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}%
_{L}\left(  t_{1}\right)  \right)  \mathbb{I}\right\}  & \mathbf{z}%
_{i}^{\dagger}\left\{  \mathbb{F}\left(  t_{1}\right)  -\mathcal{F}\left(
\mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}_{L}\left(  t_{1}\right)
\right)  \mathbb{I}\right\}
\end{array}
\right) \nonumber\\
&  \qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\times\left(
\begin{array}
[c]{c}%
i\mathbf{P}_{\mathbf{\bar{z}}_{i}}^{\bot}\left\{  \mathbb{\bar{G}}\left(
t_{2}\right)  -\mathcal{G}\left(  \mathrm{\bar{z}}_{L}\left(  t_{2}\right)
\mathrm{,z}_{L}\left(  t_{2}\right)  \right)  \mathbb{I}\right\}
\mathbf{\bar{z}}_{i}\\
-i\mathbf{P}_{\mathbf{z}_{i}}^{\bot}\left\{  \mathbb{G}\left(  t_{2}\right)
-\mathcal{G}\left(  \mathrm{\bar{z}}_{L}\left(  t_{2}\right)  \mathrm{,z}%
_{L}\left(  t_{2}\right)  \right)  \mathbb{I}\right\}  \mathbf{z}_{i}%
\end{array}
\right)  .
\end{align}
Noting that
\begin{equation}
\mathbf{z}^{\dagger}\left\{  \mathbb{X}\left(  t\right)  -\mathcal{X}\left(
t\right)  \right\}  \mathbf{z=z}^{\dagger}\mathbb{X}\left(  t\right)
\mathbf{z}-\mathcal{X}\left(  t\right)  \mathbf{z}^{\dagger}\mathbf{z}%
=\mathcal{X}\left(  t\right)  \mathbf{z}^{\dagger}\mathbf{z-}\mathcal{X}%
\left(  t\right)  \mathbf{z}^{\dagger}\mathbf{z=0}%
\end{equation}
for \emph{any }$\mathbf{z}$ shows that $\left\{  \mathbb{X}\left(  t\right)
-\mathcal{X}\left(  t\right)  \right\}  \mathbf{z}_{i}\mathbf{\,\bot\,z}_{i}$
and $\left\{  \mathbb{\bar{X}}\left(  t\right)  -\mathcal{X}\left(  t\right)
\right\}  \mathbf{\bar{z}}_{i}\mathbf{\,\bot\,\bar{z}}_{i}$ and we obtain
\begin{align}
&  \left\{  f\left(  t_{1}\right)  ,g\left(  t_{2}\right)  \right\}  =\frac
{1}{\left(  \mathbf{z}_{i}^{\dagger}\mathbf{z}_{i}\right)  ^{3}}\left(
\begin{array}
[c]{cc}%
\mathbf{\bar{z}}_{i}^{\dagger}\left\{  \mathbb{\bar{F}}\left(  t_{1}\right)
-\mathcal{F}\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}%
_{L}\left(  t_{1}\right)  \right)  \mathbb{I}\right\}  & \mathbf{z}%
_{i}^{\dagger}\left\{  \mathbb{F}\left(  t_{1}\right)  -\mathcal{F}\left(
\mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}_{L}\left(  t_{1}\right)
\right)  \mathbb{I}\right\}
\end{array}
\right) \nonumber\\
&  \quad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad
\times\left(
\begin{array}
[c]{c}%
i\left\{  \mathbb{\bar{G}}\left(  t_{2}\right)  -\mathcal{G}\left(
\mathrm{\bar{z}}_{L}\left(  t_{2}\right)  \mathrm{,z}_{L}\left(  t_{2}\right)
\right)  \mathbb{I}\right\}  \mathbf{\bar{z}}_{i}\\
-i\left\{  \mathbb{G}\left(  t_{2}\right)  -\mathcal{G}\left(  \mathrm{\bar
{z}}_{L}\left(  t_{2}\right)  \mathrm{,z}_{L}\left(  t_{2}\right)  \right)
\mathbb{I}\right\}  \mathbf{z}_{i}%
\end{array}
\right)  ,
\end{align}
which gives
\begin{align}
&  \left\{  f\left(  t_{1}\right)  ,g\left(  t_{2}\right)  \right\}
=\nonumber\\
&  \frac{1}{\left(  \mathbf{z}_{i}^{\dagger}\mathbf{z}_{i}\right)  ^{3}%
}\left\{  i\mathbf{\bar{z}}_{i}^{\dagger}\left\{  \mathbb{\bar{F}}\left(
t_{1}\right)  -\mathcal{F}\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)
\mathrm{,z}_{L}\left(  t_{1}\right)  \right)  \mathbb{I}\right\}  \left\{
\mathbb{\bar{G}}\left(  t_{2}\right)  -\mathcal{G}\left(  \mathrm{\bar{z}}%
_{L}\left(  t_{2}\right)  \mathrm{,z}_{L}\left(  t_{2}\right)  \right)
\mathbb{I}\right\}  \mathbf{\bar{z}}_{i}\right. \nonumber\\
&  \qquad\qquad\qquad\qquad\left.  -i\mathbf{z}_{i}^{\dagger}\left\{
\mathbb{\bar{F}}\left(  t_{1}\right)  -\mathcal{F}\left(  \mathrm{\bar{z}}%
_{L}\left(  t_{1}\right)  \mathrm{,z}_{L}\left(  t_{1}\right)  \right)
\mathbb{I}\right\}  \left\{  \mathbb{\bar{G}}\left(  t_{2}\right)
-\mathcal{G}\left(  \mathrm{\bar{z}}_{L}\left(  t_{2}\right)  \mathrm{,z}%
_{L}\left(  t_{2}\right)  \right)  \mathbb{I}\right\}  \mathbf{z}_{i}\right\}
\nonumber\\
&  =\frac{1}{\left(  \mathbf{z}_{i}^{\dagger}\mathbf{z}_{i}\right)  ^{3}%
}\left\{  i\mathbf{\bar{z}}_{i}^{\dagger}\left\{  \mathbb{\bar{F}}\left(
t_{1}\right)  \mathbb{\bar{G}}\left(  t_{2}\right)  -\mathcal{G}\left(
\mathrm{\bar{z}}\left(  t_{2}\right)  \mathrm{,z}\left(  t_{2}\right)
\right)  \mathbb{\bar{F}}\left(  t_{1}\right)  -\mathcal{F}\left(
\mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}_{L}\left(  t_{1}\right)
\right)  \mathbb{\bar{G}}\left(  t_{2}\right)  \right.  \right. \nonumber\\
&  \qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\left.
+\mathcal{F}\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)  \mathrm{,z}%
_{L}\left(  t_{1}\right)  \right)  \mathcal{G}\left(  \mathrm{\bar{z}}%
_{L}\left(  t_{2}\right)  \mathrm{,z}_{L}\left(  t_{2}\right)  \right)
\mathbb{I}\right\}  \mathbf{\bar{z}}_{i}\nonumber\\
&  -i\mathbf{z}_{i}^{\dagger}\left\{  \mathbb{F}\left(  t_{1}\right)
\mathbb{G}\left(  t_{2}\right)  -\mathcal{G}\left(  \mathrm{\bar{z}}\left(
t_{2}\right)  \mathrm{,z}\left(  t_{2}\right)  \right)  \mathbb{F}\left(
t_{1}\right)  -\mathcal{F}\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)
\mathrm{,z}_{L}\left(  t_{1}\right)  \right)  \mathbb{G}\left(  t_{2}\right)
\right. \nonumber\\
&  \qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\left.
\left.  +\mathcal{F}\left(  \mathrm{\bar{z}}_{L}\left(  t_{1}\right)
\mathrm{,z}_{L}\left(  t_{1}\right)  \right)  \mathcal{G}\left(
\mathrm{\bar{z}}_{L}\left(  t_{2}\right)  \mathrm{,z}_{L}\left(  t_{2}\right)
\right)  \mathbb{I}\right\}  \mathbf{z}_{i}\right\}
\end{align}
As the matrices $\mathbb{F}\left(  t_{1}\right)  $ and $\mathbb{G}\left(
t_{2}\right)  $ are hermitian
\begin{align}
\mathbf{\bar{z}}_{i}^{\dagger}\mathbb{\bar{X}}\left(  t\right)  \mathbf{\bar
{z}}_{i}  &  =\sum_{jk}z_{ij}\bar{X}_{jk}\left(  t\right)  \bar{z}%
_{ik}\nonumber\\
\mathbf{z}_{i}^{\dagger}\mathbb{X}\left(  t\right)  \mathbf{z}_{i}  &
=\sum_{jk}\bar{z}_{ij}X_{jk}\left(  t\right)  z_{ik}=\sum_{jk}\bar{z}_{ij}%
\bar{X}_{kj}\left(  t\right)  z_{ik}\nonumber\\
&  =\sum_{jk}\bar{z}_{ik}\bar{X}_{jk}\left(  t\right)  z_{ij}=\sum_{jk}%
z_{ij}\bar{X}_{jk}\left(  t\right)  \bar{z}_{ik}%
\end{align}
this leads to
\begin{equation}
\left\{  f\left(  t_{1}\right)  ,g\left(  t_{2}\right)  \right\}  =\frac
{i}{\left(  \mathbf{z}_{i}^{\dagger}\mathbf{z}_{i}\right)  ^{3}}\left\{
\mathbf{\bar{z}}_{i}^{\dagger}\mathbb{\bar{F}}\left(  t_{1}\right)
\mathbb{\bar{G}}\left(  t_{2}\right)  \mathbf{\bar{z}}_{i}-\mathbf{z}%
_{i}^{\dagger}\mathbb{F}\left(  t_{1}\right)  \mathbb{G}\left(  t_{2}\right)
\mathbf{z}_{i}\right\}  .
\end{equation}
Noting that
\begin{equation}
\sum_{jkl}z_{ij}\bar{F}_{jk}\left(  t_{1}\right)  \bar{G}_{kl}\left(
t_{2}\right)  \bar{z}_{il}=\sum_{jkl}z_{ij}F_{kj}\left(  t_{1}\right)
G_{lk}\left(  t_{2}\right)  \bar{z}_{il}=\sum_{jkl}\bar{z}_{il}G_{lk}\left(
t_{2}\right)  F_{kj}\left(  t_{1}\right)  z_{ij}%
\end{equation}
this gives
\begin{equation}
\left\{  f\left(  t_{1}\right)  ,g\left(  t_{2}\right)  \right\}  =-\frac
{i}{\left(  \mathbf{z}_{i}^{\dagger}\mathbf{z}_{i}\right)  }\mathbf{z}%
_{i}^{\dagger}\left[  \mathbb{F}\left(  t_{1}\right)  ,\mathbb{G}\left(
t_{2}\right)  \right]  \mathbf{z}_{i}.
\end{equation}
Hence
\begin{equation}
-\frac{i}{\left(  \mathbf{z}_{i}^{\dagger}\mathbf{z}_{i}\right)  }%
\mathbf{z}_{i}^{\dagger}\left[  \mathbb{F}\left(  t_{1}\right)  ,\mathbb{G}%
\left(  t_{2}\right)  \right]  \mathbf{z}_{i}=-i\mathbf{z}_{i}^{\dagger
}\left[  \mathbb{M}_{\mathcal{F}}\left(  t_{1}\right)  ,\mathbb{M}_{G}\left(
t_{2}\right)  \right]  \mathbf{z}_{i}%
\end{equation}
which can be compared to the general form%
\begin{align}
&  \left\{  f\left(  \mathrm{\bar{z}}\left(  t_{1}\right)  \mathrm{,z}\left(
t_{1}\right)  \right)  ,g\left(  \mathrm{\bar{z}}\left(  t_{2}\right)
\mathrm{,z}\left(  t_{2}\right)  \right)  \right\} \nonumber\\
&  =\left(
\begin{array}
[c]{cc}%
\frac{\partial f\left(  t\right)  }{\partial\mathbf{\bar{z}}} & \frac{\partial
f\left(  t\right)  }{\partial\mathbf{z}}%
\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
\mathbf{0} & i\overline{\mathbf{C}\left(  \mathrm{\bar{z}}\left(  t\right)
\mathrm{,z}\left(  t\right)  \right)  }^{-1}\\
-i\mathbf{C}\left(  \mathrm{\bar{z}}\left(  t\right)  \mathrm{,z}\left(
t\right)  \right)  ^{-1} & \mathbf{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\frac{\partial g\left(  t\right)  }{\partial\mathbf{\bar{z}}}\\
\frac{\partial g\left(  t\right)  }{\partial\mathbf{z}}%
\end{array}
\right) \\
&  =i\left\{  \frac{\partial f\left(  t_{1}\right)  }{\partial\mathbf{\bar{z}%
}}\overline{\mathbf{C}\left(  \mathrm{\bar{z}}\left(  t\right)  \mathrm{,z}%
\left(  t\right)  \right)  }^{-1}\frac{\partial g\left(  t_{2}\right)
}{\partial\mathbf{z}}-\frac{\partial f\left(  t_{1}\right)  }{\partial
\mathbf{z}}\mathbf{C}\left(  \mathrm{\bar{z}}\left(  t\right)  \mathrm{,z}%
\left(  t\right)  \right)  ^{-1}\frac{\partial g\left(  t_{2}\right)
}{\partial\mathbf{\bar{z}}}\right\}  .
\end{align}%
