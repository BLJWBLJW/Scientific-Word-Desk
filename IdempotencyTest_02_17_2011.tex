\subsection{Product of Vectors}

$C=A\wedge B,$ $A\in\mathcal{B}\left(  \mathcal{H}^{1}\right)  ,B\in
\mathcal{H}^{1}$%
\begin{align}
A &  =\sum_{1\leq j<k\leq4}A_{jk}\left\vert \varphi_{j}\varphi_{k}%
\right\rangle \sum_{1\leq i\leq4}B_{i}\left\vert \varphi_{i}\right\rangle
=\frac{1}{2}\sum_{1\leq i,j,k\leq4}A_{jk}B_{i}\left\vert \varphi_{j}%
\varphi_{k}\varphi_{i}\right\rangle \nonumber\\
&  =\frac{1}{2}\left\{  \left(  A_{12}B_{3}-A_{13}B_{2}+A_{23}B_{1}%
-A_{21}B_{3}-A_{32}B_{1}-A_{31}B_{2}\right)  \left\vert 123\right\rangle
\right.  \nonumber\\
&  +\left(  A_{12}B_{4}-A_{14}B_{2}+A_{24}B_{1}-A_{21}B_{4}-A_{42}B_{1}%
-A_{41}B_{2}\right)  \left\vert 124\right\rangle \nonumber\\
&  +\left(  A_{13}B_{4}-A_{14}B_{3}+A_{34}B_{1}-A_{31}B_{4}-A_{43}B_{1}%
-A_{41}B_{3}\right)  \left\vert 134\right\rangle \nonumber\\
&  \left.  +\left(  A_{23}B_{4}-A_{24}B_{3}+A_{34}B_{2}-A_{32}B_{4}%
-A_{43}B_{2}-A_{42}B_{3}\right)  \left\vert 234\right\rangle \right\}  .
\end{align}
In more detail%
\begin{align}
&  A=\sum_{1\leq j<k\leq4}A_{jk}\left\vert \varphi_{j}\varphi_{k}\right\rangle
\sum_{1\leq i\leq4}B_{i}\left\vert \varphi_{i}\right\rangle =\frac{1}{2}%
\sum_{1\leq i,,j,k\leq4}A_{jk}B_{i}\left\vert \varphi_{j}\varphi_{k}%
\varphi_{i}\right\rangle \nonumber\\
&  =\frac{1}{2}\left(  A_{12}B_{3}\left\vert 123\right\rangle +A_{13}%
B_{2}\left\vert 132\right\rangle +A_{23}B_{1}\left\vert 231\right\rangle
\right)  =\quad\left(  A_{12}B_{3}-A_{13}B_{2}+A_{23}B_{1}\right)  \left\vert
123\right\rangle \nonumber\\
&  \left.  +\right.  \frac{1}{2}\left(  A_{12}B_{4}\left\vert 124\right\rangle
+A_{14}B_{2}\left\vert 142\right\rangle +A_{24}B_{1}\left\vert
241\right\rangle \right)  =\left(  A_{12}B_{4}-A_{14}B_{2}+A_{24}B_{1}\right)
\left\vert 124\right\rangle \nonumber\\
&  \left.  +\right.  \frac{1}{2}\left(  A_{13}B_{4}\left\vert 134\right\rangle
+A_{14}B_{3}\left\vert 143\right\rangle +A_{34}B_{1}\left\vert
341\right\rangle \right)  =\left(  A_{13}B_{4}-A_{14}B_{3}+A_{34}B_{1}\right)
\left\vert 134\right\rangle \nonumber\\
&  \left.  +\right.  \frac{1}{2}\left(  A_{23}B_{4}\left\vert 234\right\rangle
+A_{24}B_{3}\left\vert 243\right\rangle +A_{34}B_{2}\left\vert
342\right\rangle \right)  =\left(  A_{23}B_{4}-A_{24}B_{3}+A_{34}B_{2}\right)
\left\vert 234\right\rangle
\end{align}
and the coefficients of the unordered basis elements \emph{are antisymmetric.
}

The normalization of the basis elements is given by
\begin{equation}

}\left\vert \varphi_{j}\varphi_{k}\right\rangle \wedge\left\vert \varphi
_{i}\right\rangle .
\end{equation}
