\subsection{Second Derivatives of the 2N-Electron Unnormalized AGP State }

\subsubsection{Calculation of the Matrix Elements $\left\langle
b_{p_{1}}^{\dagger }b_{q_{1}}^{\dagger }G^{\dagger N-1}\protect\phi %
|Hb_{p_{2}}^{\dagger }b_{q_{2}}^{\dagger }G^{\dagger N-1}\protect\phi %
\right\rangle $}

\begin{itemize}
\item $\left| p_{1}\right| \neq \left| q_{1}\right| ;\quad \left|
p_{2}\right| \neq \left| q_{2}\right| $
\end{itemize}

These subdivide into three cases $\left( p_{1},q_{1}\right) \neq \left(
p_{2},q_{2}\right) $; $\left( p_{1},q_{1}\right) $ and $\left(
p_{2},q_{2}\right) $ have one common value and $\left( p_{1},q_{1}\right)
=\left( p_{2},q_{2}\right) $. The first one with no common indices produces

\begin{equation}
\left\langle b_{p_{1}}^{\dagger }b_{q_{1}}^{\dagger }G^{\dagger N-1}\phi
|Hb_{p_{2}}^{\dagger }b_{q_{2}}^{\dagger }G^{\dagger N-1}\phi \right\rangle
=\left\{ \left( N-1\right) !\right\} ^{2}\left\langle
p_{1}q_{1}||p_{2}q_{2}\right\rangle S_{N-1}\left( \hat{p}_{1}\hat{q}_{1}\hat{%
p}_{2}\hat{q}_{2}\right)
\end{equation}%
the next one, with one common index, produces

\begin{eqnarray}
&&\left\langle b_{p_{1}}^{\dagger }b_{q_{1}}^{\dagger }G^{\dagger N-1}\phi
|Hb_{p_{1}}^{\dagger }b_{q_{2}}^{\dagger }G^{\dagger N-1}\phi \right\rangle
=\left\{ \left( N-1\right) !\right\} ^{2}\times  \nonumber \\
&&\left\{ h_{q_{1}q_{2}}S_{N-1}\left( \hat{p}_{1}\hat{q}_{1}\hat{q}%
_{2}\right) \right. +\sum_{l\neq \left| q_{1}\right| \neq \left|
q_{2}\right| =1}^{s}n_{l}S_{N-2}\left( \hat{p}_{1}\hat{q}_{1}\hat{q}_{2}\hat{%
l}\right) \left\{ \left\langle q_{1}l||q_{2}l\right\rangle +\left\langle
q_{1}l+s||q_{2}l+s\right\rangle \right\}  \nonumber \\
&&\qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad
\quad \qquad \left. +\left\langle p_{1}q_{1}||p_{1}q_{2}\right\rangle
S_{N-1}\left( \hat{p}_{1}\hat{q}_{1}\hat{q}_{2}\right) \right\}
\end{eqnarray}%
and the third one, with the pairs of indices identical, produces 
\begin{eqnarray}
&&\left\langle b_{p}^{\dagger }b_{q}^{\dagger }G^{\dagger N-1}\phi
|Hb_{p}^{\dagger }b_{q}^{\dagger }G^{\dagger N-1}\phi \right\rangle =\left\{
\left( N-1\right) !\right\} ^{2}\times  \nonumber \\
&&\left\{ S_{N-1}\left( \hat{p}\hat{q}\right) +\right. \left\{
h_{pp}+h_{qq}\right\} S_{N-1}\left( \hat{p}\hat{q}\right) +\sum_{i\neq
\left| p\right| ,\left| q\right| =1}^{s}\left\{ h_{ii}+h_{i+si+s}\right\}
n_{j}S_{N-2}\left( \hat{\imath}\hat{p}\hat{q}\right)  \nonumber \\
&&\sum_{i\neq \left| p\right| ,\left| q\right| =1}^{s}n_{i}S_{N-1}\left( 
\hat{\imath}\hat{p}\hat{q}\right) \left\langle ii+s||ii+s\right\rangle
+\sum_{i\neq \left| p\right| ,\left| q\right| =1}^{s}\zeta _{i}^{\ast }\zeta
_{j}S_{N-1}\left( \hat{\imath}\hat{p}\hat{q}\right) \left\langle
ii+s||ii+s\right\rangle  \nonumber \\
&&\sum_{i\neq \left| p\right| ,\left| q\right| =1}^{s}n_{i}S_{N-2}\left( 
\hat{\imath}\hat{p}\hat{q}\right) \left\{ \left\langle ip||ip\right\rangle
+\left\langle i+sp||i+sp\right\rangle +\left\langle iq||iq\right\rangle
+\left\langle i+sq||i+sq\right\rangle \right\}  \nonumber \\
&&\qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad
\qquad \quad +\left. S_{N-1}(\hat{p}\hat{q})\left\langle pq||pq\right\rangle
\right\}
\end{eqnarray}

\begin{itemize}

p_{2}\right| =\left| q_{2}\right| =\left| p\right| $. There are three

_{q_{1}}\left\langle p_{1}q_{1}||p_{1}p_{1}+s\right\rangle +\left\{ \zeta

\section{Computation of Second Derivatives}

\subsection{{\it Calculation of} $\left\langle b_{p_{1}}^{\dagger
}b_{q_{1}}^{\dagger }G^{\dagger N-1}\protect\phi |Hb_{p_{2}}^{\dagger
}b_{q_{2}}^{\dagger }G^{\dagger N-1}\protect\phi \right\rangle $}

This case can be further subdivided into three cases $\left(
p_{1},q_{1}\right) \neq \left( p_{2},q_{2}\right) $; $\left(
p_{1},q_{1}\right) $ and $\left( p_{2},q_{2}\right) $ have one common value
and $\left( p_{1},q_{1}\right) =\left( p_{2},q_{2}\right) $. We first deal
with

\subsubsection{{\it No common value }$\left( p_{1},q_{1}\right) \neq \left(
p_{2},q_{2}\right) $}

In this case there are no $h_{0}$ or $h_{1}$ terms, the two electron term is
calculated from 
\begin{eqnarray}
&&\sum \zeta _{j_{2}}^{\ast }\cdots \zeta _{j_{N}}^{\ast }\zeta
_{k_{2}}\cdots \zeta _{k_{N}}  \nonumber \\
&&\qquad \left\langle \eta _{p_{1}}\eta _{q_{1}}\eta _{j_{2}}\eta
_{j_{2}+s}\cdots \eta _{j_{N}\,}\eta _{j_{N}+s}\left| \frac{1}{4}%
\sum_{i,j,k,l=1}^{r}\left\langle ij||kl\right\rangle b_{i}^{\dagger
}b_{j}^{\dagger }b_{l}b_{k}\eta _{p_{2}}\eta _{q_{2}}\eta _{k_{2}\,}\eta
_{k_{2}+s}\cdots \eta _{k_{N}\,}\eta _{k_{N}+s}\right. \right\rangle 
\nonumber \\
&&\qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad _{1\leq
j_{2},\cdots ,j_{N}\leq s;\,1\leq k_{2},\cdots ,k_{N}\leq
s;\,\,\,j_{m},\,k_{m}\neq p_{1},q_{1},p_{2},q_{2}\quad 2\leq m\leq N}
\end{eqnarray}%
which equals 
\begin{eqnarray}
&&\sum \zeta _{j_{2}}^{\ast }\cdots \zeta _{j_{N}}^{\ast }\zeta
_{j_{2}}\cdots \zeta _{j_{N}}  \nonumber \\
&&\qquad \left\langle \eta _{p_{1}}\eta _{q_{1}}\eta _{j_{2}}\eta
_{j_{2}+s}\cdots \eta _{j_{N}\,}\eta _{j_{N}+s}\left| \frac{1}{4}%
\left\langle p_{1}q_{1}||p_{2}q_{2}\right\rangle b_{p_{1}}^{\dagger
}b_{q_{1}}^{\dagger }b_{q_{2}}b_{p_{2}}\eta _{p_{2}}\eta _{q_{2}}\eta
_{j_{2}\,}\eta _{j_{2}+s}\cdots \eta _{j_{N}\,}\eta _{j_{N}+s}\right.
\right\rangle  \nonumber \\
&&\qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad \qquad
\qquad _{1\leq j_{2},\cdots ,j_{N}\leq s;\,\,\,\,j_{m}\neq
p_{1},q_{1},p_{2},q_{2}\quad 2\leq m\leq N}
\end{eqnarray}%
and after summation gives the derivative 
\begin{equation}
\left\langle b_{p_{1}}^{\dagger }b_{q_{1}}^{\dagger }G^{\dagger N-1}\phi
|Hb_{p_{2}}^{\dagger }b_{q_{2}}^{\dagger }G^{\dagger N-1}\phi \right\rangle
=\left\{ \left( N-1\right) !\right\} ^{2}\left\langle
p_{1}q_{1}||p_{2}q_{2}\right\rangle S_{N-1}\left( \hat{p}_{1}\hat{q}_{1}\hat{%
p}_{2}\hat{q}_{2}\right)
\end{equation}

\subsubsection{$\left( p_{1},q_{1}\right) $ {\it and} $\left(
p_{2},q_{2}\right) $ {\it have one common value}}

We now consider the case where when two of the indices are equal. We still
do not get a $h_{0}$ term. If we let $p_{1}=p_{2}$ the one electron terms is
obtained by noting that $b_{i}^{\dagger }b_{j}$ must annihilate the $q_{2}$
from the bra and $q_{2}$ from the ket as can be seen from 
\begin{eqnarray}
&&\sum \zeta _{j_{2}}^{\ast }\cdots \zeta _{j_{N}}^{\ast }\zeta
_{k_{2}}\cdots \zeta _{k_{N}}  \nonumber \\
&&\qquad \left\langle \eta _{p_{1}}\eta _{q_{1}}\eta _{j_{2}}\eta
