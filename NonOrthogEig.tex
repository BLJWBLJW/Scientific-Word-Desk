\section{Non Orthogonal Eigenvalue Problem}

Let
\begin{align}
\mathbf{A} &  =\mathbf{A}^{\dagger}\\
\mathbf{\Delta} &  \mathbf{>0}%
\end{align}
and consider%

\begin{equation}
\mathbf{\Delta}^{-1}\mathbf{A}\mathsf{v}_{j}=\lambda_{j}\mathsf{v}_{j}%
\end{equation}
gives%
\begin{equation}
\mathbf{\Delta}^{-\frac{1}{2}}\mathbf{A\Delta}^{-\frac{1}{2}}\mathbf{\Delta
}^{\frac{1}{2}}\mathsf{v}_{j}=\lambda_{j}\mathbf{\Delta}^{\frac{1}{2}%
}\mathsf{v}_{j}%
\end{equation}
as
\begin{equation}
\mathbf{\Delta}^{-\frac{1}{2}}\mathbf{A\Delta}^{-\frac{1}{2}}=\left(
\mathbf{\Delta}^{-\frac{1}{2}}\mathbf{A\Delta}^{-\frac{1}{2}}\right)
^{\dagger}%
\end{equation}
the eigenvectors satisfy%
\begin{equation}
\mathbf{v}^{\dagger}\mathbf{\Delta v=I}%
\end{equation}
which implies that
\begin{equation}
\mathbf{\Delta}=\sum_{j}\mathsf{v}_{j}\mathsf{v}_{j}^{\dagger}.
\end{equation}
We also have
\begin{equation}
\mathbf{\Delta}^{-\frac{1}{2}}\mathbf{A\Delta}^{-\frac{1}{2}}=\sum_{j}%
\lambda_{j}\left(  \mathbf{\Delta}^{\frac{1}{2}}\mathsf{v}_{j}\right)  \left(
\mathbf{\Delta}^{\frac{1}{2}}\mathsf{v}_{j}\right)  ^{\dagger}=\sum_{j}%
\lambda_{j}\mathbf{\Delta}^{\frac{1}{2}}\mathsf{v}_{j}\mathsf{v}_{j}^{\dagger
}\mathbf{\Delta}^{\frac{1}{2}}%
\end{equation}
which gives
\begin{equation}
\mathbf{A=}\sum_{j}\lambda_{j}\mathbf{\Delta}\mathsf{v}_{j}\mathsf{v}%
_{j}^{\dagger}\mathbf{\Delta}%
\end{equation}
and
\begin{equation}
\mathbf{A-\Delta=}\sum_{j}\left\{  \lambda_{j}\mathbf{\Delta}\mathsf{v}%
_{j}\mathsf{v}_{j}^{\dagger}\mathbf{\Delta-}\mathsf{v}_{j}\mathsf{v}%
_{j}^{\dagger}\right\}
\end{equation}
