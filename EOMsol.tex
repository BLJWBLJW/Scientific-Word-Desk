\section{Quadratic Approx.}

The EOM is
\begin{equation}
\left(
\begin{array}
[c]{c}%
\overline{\mathrm{\dot{z}}}\left(  t\right)  \\
\mathrm{\dot{z}}\left(  t\right)
\end{array}
\right)  =\left(
\begin{array}
[c]{cc}%
\mathbf{0} & \overline{\mathbf{C}\left(  \mathrm{\bar{z}}\left(  t\right)
,\mathrm{z}\left(  t\right)  \right)  }^{-1}\\
-i\mathbf{C}\left(  \mathrm{\bar{z}}\left(  t\right)  ,\mathrm{z}\left(
t\right)  \right)  ^{-1} & \mathbf{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\frac{\partial\mathcal{E}_{0}\left(  \mathrm{\bar{z}}\left(  t\right)
,\mathrm{z}\left(  t\right)  \right)  }{\partial\mathbf{\bar{z}}}\\
\frac{\partial\mathcal{E}_{0}\left(  \mathrm{\bar{z}}\left(  t\right)
,\mathrm{z}\left(  t\right)  \right)  }{\partial\mathbf{z}}%
\end{array}
\right)  .
\end{equation}
In general $\mathcal{E}_{0}$ can be expanded as %

\begin{equation}
\mathcal{E}_{0}\left(  \mathbf{\bar{z}},\mathbf{z}\right)  =\mathcal{E}%
_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)
+D^{1}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma
}_{0}\right)  \left(  \mathbf{\bar{z}-\bar{\varsigma}}_{0},\mathbf{z-\varsigma
}_{0}\right)  +\frac{1}{2}D^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}%
}_{0},\mathbf{\varsigma}_{0}\right)  \left(  \mathbf{\bar{z}-\bar{\varsigma}%
}_{0},\mathbf{z-\varsigma}_{0}\right)  +\cdots,\label{enexp}%
\end{equation}
where
\begin{equation}
D_{\mathbf{\varsigma}}^{1}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  \left(  \mathbf{\bar{z}-\bar{\varsigma}%
}_{0},\mathbf{z-\varsigma}_{0}\right)  =\left(
\begin{array}
[c]{cc}%
\frac{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0}%
,\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}} & \frac
{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma
}_{0}\right)  }{\partial\mathbf{z}}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\mathbf{\bar{z}-\bar{\varsigma}}_{0}\\
\mathbf{z-\varsigma}_{0}%
\end{array}
\right)
\end{equation}
and
\begin{equation}
D_{\mathbf{\varsigma}}^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  \left(  \mathbf{\bar{z}-\bar{z}}%
_{0},\mathbf{z-z}_{0}\right)  =\left(
\begin{array}
[c]{cc}%
\mathbf{\bar{z}-\bar{\varsigma}}_{0} & \mathbf{z-\varsigma}_{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}%
}\partial\mathbf{z}}\\
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}%
\partial\mathbf{z}}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\mathbf{\bar{z}-\bar{\varsigma}}_{0}\\
\mathbf{z-\varsigma}_{0}%
\end{array}
\right)  .
\end{equation}
If $\mathcal{E}_{0}\left(  \mathbf{\bar{z}},\mathbf{z}\right)  $ is quadratic
or we make a quadratic approximation then
\begin{equation}
\mathcal{E}_{L}\left(  \mathbf{\bar{z}},\mathbf{z}\right)  =\mathcal{E}%
_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)
+D_{\mathbf{\varsigma}}^{1}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  \left(  \mathbf{\bar{z}-\bar{\varsigma}%
}_{0},\mathbf{z-\varsigma}_{0}\right)  +\frac{1}{2}D_{\mathbf{\varsigma}}%
^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  \left(  \mathbf{\bar{z}-\bar{\varsigma}}_{0},\mathbf{z-\varsigma
}_{0}\right)
\end{equation}
leading to
\begin{equation}
D_{\mathbf{z}}^{1}\mathcal{E}_{L}\left(  \mathbf{\bar{z}}_{0},\mathbf{z}%
_{0}\right)  =\left(
\begin{array}
[c]{c}%
\frac{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0}%
,\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}}\\
\frac{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0}%
,\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}}%
\end{array}
\right)  +\left(
\begin{array}
[c]{cc}%
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}%
}\partial\mathbf{z}}\\
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}%
\partial\mathbf{z}}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\mathbf{\bar{z}-\bar{\varsigma}}_{0}\\
\mathbf{z-\varsigma}_{0}%
\end{array}
\right)  .
\end{equation}
Then the linear approximation to the EOM becomes
\begin{equation}
\left(
\begin{array}
[c]{c}%
\overline{\mathrm{\dot{z}}}_{L}\left(  t\right)  \\
\mathrm{\dot{z}}_{L}\left(  t\right)
\end{array}
\right)  =\left(
\begin{array}
[c]{cc}%
\mathbf{0} & \overline{\mathbf{C}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }^{-1}\\
-i\mathbf{C}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  ^{-1} & \mathbf{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\frac{\partial\mathcal{E}_{L}\left(  \mathrm{\bar{z}}_{L}\left(  t\right)
,\mathrm{z}_{L}\left(  t\right)  \right)  }{\partial\mathbf{\bar{z}}}\\
\frac{\partial\mathcal{E}_{L}\left(  \mathrm{\bar{z}}_{L}\left(  t\right)
,\mathrm{z}_{L}\left(  t\right)  \right)  }{\partial\mathbf{z}}%
\end{array}
\right)
\end{equation}
giving
\begin{align}
\left(
\begin{array}
[c]{c}%
\overline{\mathrm{\dot{z}}}_{L}\left(  t\right)  \\
\mathrm{\dot{z}}_{L}\left(  t\right)
\end{array}
\right)    & =\left(
\begin{array}
[c]{cc}%
\mathbf{0} & \overline{\mathbf{C}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }^{-1}\\
-i\mathbf{C}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  ^{-1} & \mathbf{0}%
\end{array}
\right)  \times\nonumber\\
& \left(
\begin{array}
[c]{c}%
\frac{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0}%
,\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}}\\
\frac{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0}%
,\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}}%
\end{array}
\right)  +\left(
\begin{array}
[c]{cc}%
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}%
}\partial\mathbf{z}}\\
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}%
\partial\mathbf{z}}%
\end{array}
\right)  \left(
\begin{array}
[c]{c}%
\overline{\mathrm{z}}_{0}\left(  t\right)  \mathbf{-\bar{\varsigma}}_{0}\\
\mathrm{z}_{L}\left(  t\right)  \mathbf{-\varsigma}_{0}%
\end{array}
\right)
\end{align}
which we can abbreviate to
\begin{equation}
\mathrm{\dot{X}}\left(  t\right)  \equiv C^{-1}D_{\mathbf{\varsigma}}%
^{1}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  +C^{-1}D_{\mathbf{\varsigma}}^{2}\mathcal{E}_{0}\left(
\mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  \mathrm{X}\left(
t\right)  ,
\end{equation}
leading to
\begin{equation}
\mathrm{\dot{X}}\left(  t\right)  \equiv C^{-1}D_{\mathbf{\varsigma}}%
^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  \left\{  D_{\mathbf{\varsigma}}^{2}\mathcal{E}_{0}\left(
\mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  ^{-1}%
D_{\mathbf{\varsigma}}^{1}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  +\mathrm{X}\left(  t\right)  \right\}  .
\end{equation}
We can then make the substitution
\begin{equation}
\mathrm{Y}\left(  t\right)  =D_{\mathbf{\varsigma}}^{2}\mathcal{E}_{0}\left(
\mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  ^{-1}%
D_{\mathbf{\varsigma}}^{1}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  +\mathrm{X}\left(  t\right)
\end{equation}
obtaining
\begin{equation}
\mathrm{\dot{Y}}\left(  t\right)  =C^{-1}D_{\mathbf{\varsigma}}^{2}%
\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  \mathrm{Y}\left(  t\right)
\end{equation}
which has the solution
\begin{equation}
\mathrm{Y}\left(  t\right)  =\left\{  \exp C^{-1}D_{\mathbf{\varsigma}}%
^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  t\right\}  \mathrm{Y}\left(  0\right)  .
\end{equation}
This gives
\begin{equation}
D_{\mathbf{\varsigma}}^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  ^{-1}D_{\mathbf{\varsigma}}^{1}%
\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  +\mathrm{X}\left(  t\right)  =\left\{  \exp C^{-1}%
D_{\mathbf{\varsigma}}^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  t\right\}  \left\{  D_{\mathbf{\varsigma}%
}^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma
}_{0}\right)  ^{-1}D_{\mathbf{\varsigma}}^{1}\mathcal{E}_{0}\left(
\mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  +\mathrm{X}\left(
0\right)  \right\}
\end{equation}
leading to
\begin{align}
\mathrm{X}\left(  t\right)    & =\left\{  \exp C^{-1}D_{\mathbf{\varsigma}%
}^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma
}_{0}\right)  t\right\}  \left\{  D_{\mathbf{\varsigma}}^{2}\mathcal{E}%
_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)
^{-1}D_{\mathbf{\varsigma}}^{1}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}%
}_{0},\mathbf{\varsigma}_{0}\right)  +\mathrm{X}\left(  0\right)  \right\}
-D_{\mathbf{\varsigma}}^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  ^{-1}D_{\mathbf{\varsigma}}^{1}%
\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  \nonumber\\
& \equiv\left\{  \exp C^{-1}D_{\mathbf{\varsigma}}^{2}\mathcal{E}_{0}\left(
\mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  t\right\}
\left\{  \mathrm{X}\left(  0\right)  -X_{\min}\right\}  +X_{\min},
\end{align}
where
\begin{equation}
X_{\min}=-D_{\mathbf{\varsigma}}^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  ^{-1}D_{\mathbf{\varsigma}%
}^{1}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma
}_{0}\right)
\end{equation}
If we choose
\begin{equation}
\mathrm{X}\left(  0\right)  =\left(
\begin{array}
[c]{c}%
\overline{\mathrm{z}}_{0}\left(  0\right)  \mathbf{-\bar{\varsigma}}_{0}\\
\mathrm{z}_{L}\left(  0\right)  \mathbf{-\varsigma}_{0}%
\end{array}
\right)  =\mathbf{0,}%
\end{equation}
so that
\begin{equation}
\left(
\begin{array}
[c]{c}%
\overline{\mathrm{z}}_{0}\left(  0\right)  \\
\mathrm{z}_{L}\left(  0\right)
\end{array}
\right)  =\left(
\begin{array}
[c]{c}%
\mathbf{\bar{\varsigma}}_{0}\\
\mathbf{\varsigma}_{0}%
\end{array}
\right)  ,
\end{equation}
then this becomes
\begin{equation}
\mathrm{X}\left(  t\right)  =X_{\min}-\exp\left\{  C^{-1}D_{\mathbf{\varsigma
}}^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma
}_{0}\right)  t\right\}  X_{\min}.
\end{equation}
and
\begin{align}
& \left(
\begin{array}
[c]{c}%
\overline{\mathrm{z}}_{L}\left(  t\right)  \\
\mathrm{z}_{L}\left(  t\right)
\end{array}
\right)  =\nonumber\\
& \qquad\qquad\left\{  \exp\left(
\begin{array}
[c]{cc}%
\mathbf{0} & \overline{\mathbf{C}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }^{-1}\\
-i\mathbf{C}\left(  \mathbf{\bar{\varsigma}}_{0},\mathbf{\varsigma}%
_{0}\right)  ^{-1} & \mathbf{0}%
\end{array}
\right)  \left(
\begin{array}
[c]{cc}%
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}%
}\partial\mathbf{z}}\\
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}%
\partial\mathbf{z}}%
\end{array}
\right)  t\right\}  \nonumber\\
& \qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad\times\left(
\begin{array}
[c]{cc}%
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}%
}\partial\mathbf{z}}\\
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}%
\partial\mathbf{z}}%
\end{array}
\right)  ^{-1}\left(
\begin{array}
[c]{c}%
\frac{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0}%
,\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}}\\
\frac{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0}%
,\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}}%
\end{array}
\right)  \nonumber\\
& \qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad-\left(
\begin{array}
[c]{cc}%
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}%
}\partial\mathbf{z}}\\
\frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}%
_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}\partial
\mathbf{\bar{z}}} & \frac{\partial^{2}\mathcal{E}_{0}\left(  \mathbf{\bar
{\varsigma}}_{0},\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}%
\partial\mathbf{z}}%
\end{array}
\right)  ^{-1}\left(
\begin{array}
[c]{c}%
\frac{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0}%
,\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{\bar{z}}}\\
\frac{\partial\mathcal{E}_{0}\left(  \mathbf{\bar{\varsigma}}_{0}%
,\mathbf{\varsigma}_{0}\right)  }{\partial\mathbf{z}}%
\end{array}
\right)  +\left(
\begin{array}
[c]{c}%
\mathbf{\bar{\varsigma}}_{0}\\
\mathbf{\varsigma}_{0}%
\end{array}
\right)  .
\end{align}
This shows that
\begin{align}
\mathrm{\dot{X}}\left(  t\right)    & =\left(
\begin{array}
[c]{c}%
\overline{\mathrm{\dot{z}}}_{L}\left(  t\right)  \mathbf{-\bar{\varsigma}}%
_{0}\\
\mathrm{\dot{z}}_{L}\left(  t\right)  \mathbf{-\varsigma}_{0}%
\end{array}
\right)  =\mathbf{0}\\
& \Rightarrow\left(
\begin{array}
[c]{c}%
\overline{\mathrm{\dot{z}}}_{L}\left(  t\right)  \\
\mathrm{\dot{z}}_{L}\left(  t\right)
\end{array}
\right)  =\left(
\begin{array}
[c]{c}%
\mathbf{\bar{\varsigma}}_{0}\\
\mathbf{\varsigma}_{0}%
\end{array}
\right)
\end{align}
at a critical point $\left(
\begin{array}
[c]{c}%
\mathbf{\bar{\varsigma}}_{0}\\
\mathbf{\varsigma}_{0}%
\end{array}
\right)  $ of $\mathcal{E}_{0}.$
