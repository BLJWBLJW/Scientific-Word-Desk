\section{Second Quantization}

In the following $\mathcal{H}^{1}$ is a $r$ dimensionsal $\mathbb{C}$-Hilbert
space with a conb $\left\{  \left.  \varphi_{j}\right\vert 1\leq j\leq
r\right\}  .$

\begin{notation}%
\begin{equation}
\left\vert \varphi_{j_{1}}\cdots\varphi_{j_{N}}\right\rangle =\sqrt
{N!}\left\vert \varphi_{j_{1}}\wedge\cdots\wedge\varphi_{j_{N}}\right\rangle ,
\end{equation}
so that
\begin{equation}
\left\Vert \left\vert \varphi_{j_{1}}\cdots\varphi_{j_{N}}\right\rangle
\right\Vert _{\mathcal{H}_{\mathcal{F}}}^{2}=\left\langle \left.
\varphi_{j_{1}}\cdots\varphi_{j_{N}}\right\vert \varphi_{j_{1}}\cdots
\varphi_{j_{N}}\right\rangle =1.
\end{equation}

\end{notation}

\begin{definition}
The second quantization map
\begin{equation}
\Gamma\in\mathcal{L}\left(  \mathcal{B}\left(  \mathcal{H}_{\mathcal{F}%
}\right)  \right)
\end{equation}
is defined by
\begin{align}
&  \Gamma\left(  \left\vert \varphi_{k_{1}}\cdots\varphi_{k_{P}}\right\rangle
\left\langle \varphi_{l_{1}}\cdots\varphi_{l_{Q}}\right\vert \right)
=\sum_{0\leq R,S\leq r}\Gamma_{RS}\left(  \left\vert \varphi_{k_{1}}%
\cdots\varphi_{k_{P}}\right\rangle \left\langle \varphi_{l_{1}}\cdots
\varphi_{l_{Q}}\right\vert \right)  =\Gamma_{PQ}\left(  \left\vert
\varphi_{k_{1}}\cdots\varphi_{k_{P}}\right\rangle \left\langle \varphi_{l_{1}%
}\cdots\varphi_{l_{Q}}\right\vert \right) \\
&  =\sum_{0\leq N\leq r}\binom{N+P}{P}^{\frac{1}{2}}\binom{N+Q}{Q}^{\frac
{1}{2}}\left\vert \varphi_{k_{1}}\cdots\varphi_{k_{P}}\right\rangle
\left\langle \varphi_{l_{1}}\cdots\varphi_{l_{Q}}\right\vert \bullet
P_{\mathcal{H}^{N}}\\
&  =\sum_{0\leq N\leq r}\binom{N+P}{P}^{\frac{1}{2}}\binom{N+Q}{Q}^{\frac
{1}{2}}\Gamma_{PQ}^{N+P,N+Q}\left(  \left\vert \varphi_{k_{1}}\cdots
\varphi_{k_{P}}\right\rangle \left\langle \varphi_{l_{1}}\cdots\varphi_{l_{Q}%
}\right\vert \right) \\
&  =\sum_{0\leq N\leq r}\sqrt{\binom{N+P}{P}}\sqrt{\binom{N+Q}{Q}}\Gamma
_{PQ}^{N+P,N+Q}\left(  \left\vert \varphi_{k_{1}}\cdots\varphi_{k_{P}%
}\right\rangle \left\langle \varphi_{l_{1}}\cdots\varphi_{l_{Q}}\right\vert
\right) \\
&  =\sum_{0\leq N\leq r}\sum_{1\leq j_{1}<\cdots<j_{N}\leq r}\left\vert
\varphi_{k_{1}}\cdots\varphi_{k_{P}}\varphi_{j_{1}}\cdots\varphi_{j_{N}%
}\right\rangle \left\langle \varphi_{l_{1}}\cdots\varphi_{l_{Q}}\varphi
_{j_{1}}\cdots\varphi_{j_{N}}\right\vert ,
\end{align}
where

\begin{remark}
$\Gamma$ is a linear map it does not produce the CAR's.
\end{remark}

\begin{enumerate}
\item
\begin{gather}
a_{k_{1}}^{\dagger}\cdots a_{k_{P}}^{\dagger}a_{l_{Q}}\cdots a_{l_{1}}%
=\Gamma\left(  \left\vert \varphi_{k_{1}}\cdots\varphi_{k_{P}}\right\rangle
\left\langle \varphi_{l_{1}}\cdots\varphi_{l_{Q}}\right\vert \right) \\
1\leq k_{1}<\cdots<k_{P}\leq r,1\leq l_{1}<\cdots<l_{Q}\leq r\ ,\\
0\leq P,Q\leq r.
\end{gather}


\item $\left\{  \left.  \left\vert \varphi_{j_{1}}\cdots\varphi_{j_{N}%
}\right\rangle \right\vert 0\leq N\leq r,1\leq k_{1}<\cdots<k_{N}\leq
r\right\}  $ is a conb for $\mathcal{H}_{\mathcal{F}}.$

\item The ket-bra operators $\left\vert \varphi_{k_{1}}\cdots\varphi_{k_{P}%
}\right\rangle \left\langle \varphi_{l_{1}}\cdots\varphi_{l_{Q}}\right\vert
\in\mathcal{B}\left(  \mathcal{H}_{\mathcal{F}}\right)  $ are defined by
\begin{gather}
\left[  \overset{}{\left\vert \varphi_{k_{1}}\cdots\varphi_{k_{P}%
}\right\rangle \left\langle \varphi_{l_{1}}\cdots\varphi_{l_{Q}}\right\vert
}\right]  \left\vert \varphi_{m_{1}}\cdots\varphi_{m_{N}}\right\rangle
=\left\vert \varphi_{k_{1}}\cdots\varphi_{k_{P}}\right\rangle \left\langle
\left.  \varphi_{l_{1}}\cdots\varphi_{l_{Q}}\right\vert \varphi_{m_{1}}%
\cdots\varphi_{m_{N}}\right\rangle \\
1\leq k_{1}<\cdots<k_{P}\leq r,1\leq l_{1}<\cdots<l_{Q}\leq r,1\leq
m_{1}<\cdots<m_{N}\leq r\\
0\leq N,P,Q\leq r.
\end{gather}


\item The orthogonal projector onto the subspace $\mathcal{H}^{N}$ is given
by
\begin{equation}
P_{\mathcal{H}^{N}}=\sum_{1\leq j_{1}<\cdots<j_{N}\leq r}\left\vert
\varphi_{j_{1}}\cdots\varphi_{j_{N}}\right\rangle \left\langle \varphi_{j_{1}%
}\cdots\varphi_{j_{N}}\right\vert .
\end{equation}


\item $\mathcal{H}_{\mathcal{F}}$ is the exterior Hilbert algebra with
identity $\left\vert \phi\right\rangle $ i.e.

\begin{enumerate}
\item
\begin{equation}
\mathcal{H}_{\mathcal{F}}={\LARGE \wedge}\left(  \mathcal{F}\right)
=\bigoplus\limits_{0\leq N\leq r}\mathcal{H}^{N},
\end{equation}


\item
\begin{equation}
\mathcal{H}^{N}={\LARGE \wedge}^{N}\mathcal{H}^{1}%
\end{equation}


\item The Hilbert space is given by
\begin{equation}
\mathcal{H}^{1}=\mathbb{C}\text{-}\operatorname{linspan}\left\{  \left.
\left\vert \varphi_{j}\right\rangle \right\vert 1\leq j\leq r\right\}
\end{equation}


\item $\left\vert \phi\right\rangle $ is a vacuum vector and
\begin{equation}
\mathcal{H}^{0}=\mathbb{C}\left\vert \phi\right\rangle .
\end{equation}

\end{enumerate}

\item The cases $N=P=Q=0$ are defined by
\begin{equation}
\left\vert \varphi_{k_{1}}\cdots\varphi_{k_{0}}\right\rangle =\left\vert
\phi\right\rangle
\end{equation}


\item The product "$\bullet$"
\begin{equation}
\bullet:\mathcal{B}\left(  \mathcal{H}_{\mathcal{F}}\right)  \times
\mathcal{B}\left(  \mathcal{H}_{\mathcal{F}}\right)  \rightarrow
\mathcal{B}\left(  \mathcal{H}_{\mathcal{F}}\right)
\end{equation}
is defined by
\begin{equation}
\left\vert \Psi_{1}\right\rangle \left\langle \Phi_{1}\right\vert
\bullet\left\vert \Psi_{2}\right\rangle \left\langle \Phi_{2}\right\vert
=\left\vert \Psi_{1}\wedge\Psi_{2}\right\rangle \left\langle \Phi_{1}%
\wedge\Phi_{2}\right\vert ,\quad\Psi_{1},\Psi_{2},\Phi_{1},\Phi_{2}%
\in\mathcal{H}_{\mathcal{F}}.
\end{equation}

\end{enumerate}
\end{definition}

\begin{remark}%
\begin{equation}
a_{k_{1}}^{\dagger}a_{l_{1}}=\Gamma\left(  \left\vert \varphi_{k_{1}%
}\right\rangle \left\langle \varphi_{l_{1}}\right\vert \right)
\end{equation}

\end{remark}

\section{Majorano Operators}

\begin{definition}
\begin{enumerate}
\item Creation and annihilation operators in terms of Majorano operators
\begin{align}
a_{j}^{\dagger}  &  =\frac{1}{\sqrt{2}}\left(  x_{j}-ix_{j+r}\right) \\
a_{j}  &  =\frac{1}{\sqrt{2}}\left(  x_{j}+ix_{j+r}\right)
\end{align}


\item Majorano operators in terms of Creation and annihilation operators
\begin{align}
x_{j}  &  =\frac{1}{\sqrt{2}}\left(  a_{j}^{\dagger}+a_{j}\right) \\
x_{j+r}  &  =\frac{i}{\sqrt{2}}\left(  a_{j}^{\dagger}-a_{j}\right)  .
\end{align}

\end{enumerate}
\end{definition}

\begin{remark}
The second quantiztion map $\Gamma$ acting on ket bras to produce Majorano
operators%
\begin{align}
x_{j}  &  =\frac{1}{\sqrt{2}}\left(  \overset{}{\Gamma\left(  \left\vert
\varphi_{j}\right\rangle \left\langle \phi\right\vert \ \right)
+\Gamma\left(  \left\vert \phi\right\rangle \left\langle \varphi
_{j}\right\vert \ \right)  }\right)  =\frac{1}{\sqrt{2}}\Gamma\left(
\overset{}{\left\vert \varphi_{j}\right\rangle \left\langle \phi\right\vert
+\left\vert \phi\right\rangle \left\langle \varphi_{j}\right\vert }\right) \\
x_{j+r}  &  =\frac{i}{\sqrt{2}}\left(  \overset{}{\Gamma\left(  \left\vert
\varphi_{j}\right\rangle \left\langle \phi\right\vert \ \right)
-\Gamma\left(  \left\vert \phi\right\rangle \left\langle \varphi
_{j}\right\vert \ \right)  }\right)  =\frac{i}{\sqrt{2}}\Gamma\left(
\overset{}{\left\vert \varphi_{j}\right\rangle \left\langle \phi\right\vert
-\left\vert \phi\right\rangle \left\langle \varphi_{j}\right\vert }\right)  .
\end{align}

\end{remark}

\begin{remark}
CAR for Mjorano and second quantized operators relationships
\begin{align}
\left\{  x_{j},x_{k+r}\right\}   &  =\frac{i}{2}\left\{  \overset{}{\left(
a_{j}^{\dagger}+a_{j}\right)  ,\left(  a_{k}^{\dagger}-a_{k}\right)
}\right\}  =\frac{i}{2}\left\{  \Gamma\left(  \overset{}{\left\vert

\phi\right\rangle \left\langle \varphi_{j}\right\vert }\right)  ,\left(

\phi\right\vert -\left\vert \phi\right\rangle \left\langle \varphi
_{k}\right\vert }\right)  \right)  \right\}  =0\\

a_{j}^{\dagger}+a_{j}\right)  ,\left(  a_{k}^{\dagger}+a_{k}\right)
}\right\}  =\frac{1}{2}\left\{  \Gamma\left(  \overset{}{\left\vert

\phi\right\vert -\left\vert \phi\right\rangle \left\langle \varphi
_{k}\right\vert }\right)  \right)  \right\}  =I_{\mathcal{F}}\delta

\end{remark}

then
\begin{equation}
\left\vert g\right\rangle \left\langle g\right\vert ^{2}=\left\vert
g\right\rangle \left\langle g\right\vert
\end{equation}
and
\begin{equation}
\left\vert g\right\rangle \left\langle g\right\vert ^{\dagger}=\left\vert
g\right\rangle \left\langle g\right\vert
\end{equation}
thus $\left\vert g\right\rangle \left\langle g\right\vert $ is an orthogonal projector.
\end{remark}
