The expectation value of $T\left(  \nu,\omega\right)  ^{\dagger}\left(
\mathbf{P\bullet P}\right)  T\left(  \nu,\omega\right)  $ is given by %

\begin{align*}
&  \left\langle \Psi|T\left(  \nu,\omega\right)  ^{\dagger}\left(
\mathbf{P\bullet P}\right)  T\left(  \nu,\omega\right)  \Psi\right\rangle =\\
- &  \left\langle \left(  \mathbf{\nabla}_{j}-\mathbf{\nabla}_{j}%
\omega\right)  \nu_{N}\Psi|\left(  \mathbf{\nabla}_{j}-\mathbf{\nabla}%
_{j}\omega\right)  \nu_{N}\Psi\right\rangle \\
&  =-\left\langle \left\{  \left(  \mathbf{\nabla}_{j}\nu_{N}\right)  +\nu
_{N}\mathbf{\nabla}_{j}-\nu_{N}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)
\right\}  \Psi|\left\{  \left(  \mathbf{\nabla}_{j}\nu_{N}\right)  +\nu
_{N}\mathbf{\nabla}_{j}-\nu_{N}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)
\right\}  \Psi\right\rangle \\
&  =\left\langle \Psi|\left\{  \mathbf{\nabla}_{j}\nu_{N}\bullet
\mathbf{\nabla}_{j}\nu_{N}+\nu_{N}\left(  \mathbf{\nabla}_{j}\nu_{N}\right)
\bullet\mathbf{\nabla}_{j}-\nu_{N}\mathbf{\nabla}_{j}\nu_{N}\bullet
\mathbf{\nabla}_{j}\omega\right.  \right.  \\
&  +\nu_{N}\left(  \mathbf{\nabla}_{j}\nu_{N}\right)  \bullet\mathbf{\nabla
}_{j}+\left(  \nu_{N}\right)  ^{2}\mathbf{\nabla}_{j}^{2}-\nu_{N}%
\mathbf{\nabla}_{j}\nu_{N}\bullet\mathbf{\nabla}_{j}\omega-\left(  \nu
_{N}\right)  ^{2}\left(  \mathbf{\nabla}_{j}^{2}\omega\right)  -\nu
_{N}\mathbf{\nabla}_{j}\omega\bullet\mathbf{\nabla}_{j}\\
&  \left.  -\nu_{N}\mathbf{\nabla}_{j}\omega_{N}\bullet\mathbf{\nabla}_{j}%
\nu_{N}-\left(  \nu_{N}\right)  ^{2}\mathbf{\nabla}_{j}\omega_{N}%
\bullet\mathbf{\nabla}_{j}+\nu_{N}^{2}\mathbf{\nabla}_{j}\omega_{N}%
\bullet\mathbf{\nabla}_{j}\omega_{N}\right\}  \left.  \Psi\right\rangle
\end{align*}
we break this term into the parts
\begin{align}
& \left\langle \left(  \mathbf{\nabla}_{j}\nu_{N}\right)  \Psi|\left\{
\left(  \mathbf{\nabla}_{j}\nu_{N}\right)  +\nu_{N}\mathbf{\nabla}_{j}-\nu
_{N}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)  \right\}  \Psi\right\rangle
\nonumber\\
& =\left\langle \Psi|\left(  \mathbf{\nabla}_{j}\nu_{N}\right)  \left\{
\left(  \mathbf{\nabla}_{j}\nu_{N}\right)  +\nu_{N}\mathbf{\nabla}_{j}-\nu
_{N}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)  \right\}  \Psi\right\rangle
\nonumber\\
& =\left\langle \Psi|\left\{  \left(  \mathbf{\nabla}_{j}\nu_{N}\right)
\bullet\left(  \mathbf{\nabla}_{j}\nu_{N}\right)  +\nu_{N}\left(
\mathbf{\nabla}_{j}\nu_{N}\right)  \bullet\mathbf{\nabla}_{j}-\nu_{N}\left(
\mathbf{\nabla}_{j}\nu_{N}\right)  \bullet\left(  \mathbf{\nabla}_{j}%
\omega_{N}\right)  \right\}  \Psi\right\rangle
\end{align}%
\begin{align}
& \left\langle \nu_{N}\mathbf{\nabla}_{j}\Psi|\left\{  \left(  \mathbf{\nabla
}_{j}\nu_{N}\right)  +\nu_{N}\mathbf{\nabla}_{j}-\nu_{N}\left(  \mathbf{\nabla
}_{j}\omega_{N}\right)  \right\}  \Psi\right\rangle \nonumber\\
& =-\left\langle \Psi|\nu_{N}\mathbf{\nabla}_{j}\left\{  \left(
\mathbf{\nabla}_{j}\nu_{N}\right)  +\nu_{N}\mathbf{\nabla}_{j}-\nu_{N}\left(
\mathbf{\nabla}_{j}\omega_{N}\right)  \right\}  \Psi\right\rangle \nonumber\\
& =-\left\langle \Psi|\left\{  \nu_{N}\left(  \mathbf{\nabla}_{j}^{2}\nu
_{N}\right)  +\nu_{N}\left(  \mathbf{\nabla}_{j}\nu_{N}\right)  \bullet
\mathbf{\nabla}_{j}+\nu_{N}^{2}\mathbf{\nabla}_{j}^{2}\right.  \right.
\nonumber\\
& \left.  \left.  +\nu_{N}\left(  \mathbf{\nabla}_{j}\nu_{N}\right)
\bullet\mathbf{\nabla}_{j}-\nu_{N}\left(  \mathbf{\nabla}_{j}\nu_{N}\right)
\bullet\left(  \mathbf{\nabla}_{j}\omega_{N}\right)  -\nu_{N}^{2}\left(
\mathbf{\nabla}_{j}^{2}\omega_{N}\right)  \Psi\right\}  \right\rangle
\end{align}%
\begin{align}
& \left\langle \nu_{N}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)
\Psi|\left\{  \left(  \mathbf{\nabla}_{j}\nu_{N}\right)  +\nu_{N}%
\mathbf{\nabla}_{j}-\nu_{N}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)
\right\}  \Psi\right\rangle \nonumber\\
& =\left\langle \Psi|\nu_{N}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)
\left\{  \left(  \mathbf{\nabla}_{j}\nu_{N}\right)  +\nu_{N}\mathbf{\nabla
}_{j}-\nu_{N}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)  \right\}
\Psi\right\rangle \nonumber\\
& =\left\langle \Psi|\left\{  \nu_{N}\left(  \mathbf{\nabla}_{j}\omega
_{N}\right)  \bullet\left(  \mathbf{\nabla}_{j}\nu_{N}\right)  +\nu_{N}%
^{2}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)  \bullet\mathbf{\nabla}%
_{j}\right.  \right.  \nonumber\\
& \qquad\qquad\qquad\qquad\left.  \left.  -\nu_{N}^{2}\left(  \mathbf{\nabla
}_{j}\omega_{N}\right)  \bullet\left(  \mathbf{\nabla}_{j}\omega_{N}\right)
\right\}  \right\rangle \Psi
\end{align}
which we can collect into
\begin{align}
& \left\langle \Psi|\left\{  \left(  \mathbf{\nabla}_{j}\nu_{N}\right)
\bullet\left(  \mathbf{\nabla}_{j}\nu_{N}\right)  +\nu_{N}\left(
\mathbf{\nabla}_{j}\nu_{N}\right)  \bullet\left(  \mathbf{\nabla}_{j}%
\omega_{N}\right)  \right.  \right.  \nonumber\\
& \qquad-\nu_{N}^{2}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)
\bullet\left(  \mathbf{\nabla}_{j}\omega_{N}\right)  -\nu_{N}\left(
\mathbf{\nabla}_{j}^{2}\nu_{N}\right)  +\nu_{N}^{2}\left(  \mathbf{\nabla}%
_{j}^{2}\omega_{N}\right)  \nonumber\\
& \qquad\qquad\left.  \left.  -\nu_{N}^{2}\mathbf{\nabla}_{j}^{2}-\nu
_{N}\left(  \mathbf{\nabla}_{j}\nu_{N}\right)  \bullet\mathbf{\nabla}_{j}%
+\nu_{N}^{2}\left(  \mathbf{\nabla}_{j}\omega_{N}\right)  \bullet
\mathbf{\nabla}_{j}\right\}  \Psi\right\rangle
\end{align}
